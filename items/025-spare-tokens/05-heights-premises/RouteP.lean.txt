/-- The delete's invariant at every fuel above the fuel and the next id. -/
def DelInvA (c : Ctx) (base : View BVal) (w0 : W) (key : Key) (KO : KindOracle) (BK : NodeId)
    (v : W) : Prop :=
  ∀ F, c.p.fuel + 1 ≤ F → v.next < F → DelInv c base w0 key F KO BK v

/-- P3, up: at p.fuel + 1 it gives every larger fuel (the reading by

/-- P4: `DelStepsOK` with its obligations over an invariant I of the
consumer's choice: I holds where `DelInv` at p.fuel + 1 holds (at the calls),
and gives it back at the operation's end, where the state fits between
operations. `DelStepsOK` is the case I := `DelInv` at p.fuel + 1. -/
def DelStepsOKG (p : Params) (R : Struct BVal → Time → Prop)
    (I : List Tx → Tx → W → Key → W → Prop) : Prop :=
  ∀ ts tx, Within p (ts ++ [tx]) → R (treeRun p ts) ts.length → ∀ z top,
    lastTop (view (treeSpec p) (treeRun p ts) ts.length) = some z →
    topOf p (ctxOf p ts).view z = .ok top →
    ∀ pre op post, tx.ops = pre ++ op :: post →
    ∀ r w, (pre.mapM putKv) (ctxOf p ts) (wInit tx top) = .ok (r, w) →
    PutInv (ctxOf p ts) (viewOf (treeRun p ts)) w →
    RangeOrder (wNode p (viewOf (treeRun p ts)) w) (ts.length + 1) w.top.rootKey →
    SegFits (wNode p (viewOf (treeRun p ts)) w) (ts.length + 1) (p.fuel - 1) w.top.rootKey →
    ∀ key, op = .del key → p.merges = true →
    (∀ v, DelInv (ctxOf p ts) (viewOf (treeRun p ts)) w key (p.fuel + 1)
        (kindOracleOf p (treeRun p ts)) tx.base v → I ts tx w key v) →
    (∀ v, I ts tx w key v →
      SegFits (wNode p (viewOf (treeRun p ts)) v) (ts.length + 1) (p.fuel - 1) v.top.rootKey →
      DelInv (ctxOf p ts) (viewOf (treeRun p ts)) w key (p.fuel + 1)
        (kindOracleOf p (treeRun p ts)) tx.base v) →
    (∀ g h v, I ts tx w key v →
      WriterWP (afterDelete key g h) (ctxOf p ts) v (fun _ v' => I ts tx w key v')) →
    (∀ g h merged v, I ts tx w key v →
      WriterWP (mergeUp key g p.fuel h merged) (ctxOf p ts) v (fun _ v' => I ts tx w key v')) →
    (∀ seg v, I ts tx w key v →
      WriterWP (collapseRoot key seg p.fuel) (ctxOf p ts) v (fun _ v' => I ts tx w key v')) →
    WriterWP (putKv op) (ctxOf p ts) w (fun _ w' =>
      SegFits (wNode p (viewOf (treeRun p ts)) w') (ts.length + 1) (p.fuel - 1) w'.top.rootKey →
      ∀ key', keyWf key' = true →
        runGet p ts w' key' = if op.key == key' then op.value else runGet p ts w key')

/-- The old family is the instance at `DelInv` (no proof kept twice). -/
theorem delStepsOK_of_G (p : Params) (R : Struct BVal → Time → Prop)
    (h : DelStepsOKG p R (fun ts tx w key v => DelInv (ctxOf p ts) (viewOf (treeRun p ts)) w key
      (p.fuel + 1) (kindOracleOf p (treeRun p ts)) tx.base v)) :
    DelStepsOK p R := by
  intro ts tx hw hr z top hz htop pre op post hops r w hpre hI hRO hF key hop hm hA hMU hCR
  exact h ts tx hw hr z top hz htop pre op post hops r w hpre hI hRO hF key hop hm
    (fun _ hv => hv) (fun _ hv _ => hv) hA hMU hCR

/-- The family at `DelInvA`, which P2 discharges with no premise. -/
abbrev DelStepsOKA (p : Params) (R : Struct BVal → Time → Prop) : Prop :=
  DelStepsOKG p R (fun ts tx w key v => DelInvA (ctxOf p ts) (viewOf (treeRun p ts)) w key
    (kindOracleOf p (treeRun p ts)) tx.base v)
