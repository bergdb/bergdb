/-- When `targetGo` returns `collapse x e`, x is the path's node at depth e, an inner node, and
the node above depth e, if any, is a segment leaf. -/
theorem targetGo_collapse (c : Ctx) (base : View BVal) (hvc : ViewCoherent c base) :
    ∀ (d : Nat) (w : W), (∀ y, w.taken.contains y = false) →
      WriterWP (targetGo d) c w (fun f _ => ∀ x e, f = Fix.collapse x e →
        (∃ a xn, w.path[e]? = some (a, x) ∧ wNode c.p base w x = some xn ∧ xn.leaf = false) ∧
        (e = 0 ∨ ∃ q qn, w.path[e - 1]? = some q ∧ wNode c.p base w q.2 = some qn ∧
          qn.leaf = true)) := by
  intro d
  induction d with
  | zero => intro w _; simp only [targetGo, writerWP_pure]; intro x e h; cases h
  | succ d ih =>
    intro w ht
    rw [targetGo]
    simp only [writerWP_bind, writerWP_read]
    apply pathNode_wp
    intro e0 he0
    refine writerWP_mono (getNode_rp e0.2 c w base hvc) ?_
    intro xn v ⟨hrv, hpv, hxn'⟩
    have hxn := hxn' (ht e0.2)
    have htv : ∀ y, v.taken.contains y = false := by intro y; rw [hrv.2.2.1]; exact ht y
    have hrec : ∀ v', ReadsAs c.p base w v' → v'.path = w.path →
        WriterWP (targetGo d) c v' (fun f _ => ∀ x e, f = Fix.collapse x e →
          (∃ a xn, w.path[e]? = some (a, x) ∧ wNode c.p base w x = some xn ∧ xn.leaf = false) ∧
          (e = 0 ∨ ∃ q qn, w.path[e - 1]? = some q ∧ wNode c.p base w q.2 = some qn ∧
            qn.leaf = true)) := by
      intro v' hr hp
      have ht' : ∀ y, v'.taken.contains y = false := by intro y; rw [hr.2.2.1]; exact ht y
      refine writerWP_mono (ih v' ht') ?_
      intro f _ h x e he
      have := h x e he
      rw [hr.1, hp] at this
      exact this
    have hother : ∀ (f : Fix) v', (∀ x e, f ≠ Fix.collapse x e) →
        WriterWP (pure f : M Fix) c v' (fun f _ => ∀ x e, f = Fix.collapse x e →
          (∃ a xn, w.path[e]? = some (a, x) ∧ wNode c.p base w x = some xn ∧ xn.leaf = false) ∧
          (e = 0 ∨ ∃ q qn, w.path[e - 1]? = some q ∧ wNode c.p base w q.2 = some qn ∧
            qn.leaf = true)) := by
      intro f v' hf
      simp only [writerWP_pure]
      intro x e he
      exact absurd he (hf x e)
    split
    · simp only [writerWP_bind]; exact writerWP_throw _ _ _ _
    · cases d with
      | zero =>
        simp only [writerWP_bind, writerWP_pure]
        rw [if_pos trivial]
        by_cases hl : xn.leaf = true
        · rw [if_pos hl]
          have h00 : ¬ ((xn.cnt == 0 && 0 != 0) = true) := by simp
          rw [if_neg h00]
          exact hrec v hrv hpv
        · rw [if_neg hl]
          by_cases hc : (xn.cnt == 1) = true
          · rw [if_pos hc]
            simp only [writerWP_pure]
            intro x e he
            cases he
            exact ⟨⟨e0.1, xn, by simpa using he0, hxn, by simpa using hl⟩, Or.inl rfl⟩
          · rw [if_neg hc]; exact hrec v hrv hpv
      | succ e1 =>
        simp only [writerWP_bind, writerWP_pure]
        apply pathNode_wp
        intro e' he'
        rw [hpv] at he'
        refine writerWP_mono (getNode_rp e'.2 c v base hvc) ?_
        intro qn v2 ⟨hr2, hp2, hqn'⟩
        have hqn := hqn' (htv e'.2)
        rw [hrv.1] at hqn
        have hr02 := ReadsAs.trans _ _ _ _ _ hrv hr2
        have hp02 : v2.path = w.path := hp2.trans hpv
        by_cases hql : qn.leaf = true
        · rw [if_pos hql]
          by_cases hl : xn.leaf = true
          · rw [if_pos hl]
            by_cases hc : (xn.cnt == 0 && e1 + 1 != 0) = true
            · rw [if_pos hc]; exact hother _ v2 (fun x e h => by cases h)
            · rw [if_neg hc]; exact hrec v2 hr02 hp02
          · rw [if_neg hl]
            by_cases hc : (xn.cnt == 1) = true
            · rw [if_pos hc]
              simp only [writerWP_pure]
              intro x e he
              cases he
              exact ⟨⟨e0.1, xn, by simpa using he0, hxn, by simpa using hl⟩,
                Or.inr ⟨e', qn, by simpa using he', hqn, hql⟩⟩
            · rw [if_neg hc]; exact hrec v2 hr02 hp02
        · rw [if_neg hql]
          by_cases hlt : xn.cnt < c.p.mergeBelow
          · rw [if_pos hlt]
            by_cases hq2 : 2 ≤ qn.cnt
            · rw [if_pos hq2]; exact hother _ v2 (fun x e h => by cases h)
            · rw [if_neg hq2]
              simp only [writerWP_bind]
              rw [writerWP_note_iff]
              exact hrec _ (ReadsAs.trans _ _ _ _ _ hr02 ⟨by funext y; rfl, rfl, rfl, rfl, rfl⟩) hp02
          · rw [if_neg hlt]; exact hrec v2 hr02 hp02

theorem nex_mtCollapse (B : NodeId) (x : NodeId) (d : Nat) : KeepsNE B (Mt.collapse x d) := by
  intro E c w hv hw hn
  have hpos := hv.pos
  unfold Mt.collapse
  repeat' first
    | wpSimplify
    | neCore
    | ((with_unfolding_none guard_target = WriterWP (replaceRoot _ _ _) _ _ _)
       refine writerWP_mono (nex_replaceRoot B _ _ _ _ _ _ hv (by assumption) (by assumption) (by neBase)) ?_
       rintro _ _ ⟨_, _⟩)
    | wpSplit
    | split
    | contradiction

/-- A run of `Mt.collapse x d` from a state with x cached: the checks, a note, `replaceRoot`. -/
theorem mtCollapse_wp (x : NodeId) (d : Nat) (c : Ctx) (w : W) (n : KNode)
    (hlk : w.nodes.lookup x = some n) (post : Unit → W → Prop)
    (h : ∀ only, (n.firstTime == some c.i) = false → n.onlyLiveKlong = some only →
      n.cv? only = some 0 → ∀ s, WriterWP (replaceRoot x d only) c { w with trail := s :: w.trail } post) :
    WriterWP (Mt.collapse x d) c w post := by
  unfold Mt.collapse
  simp only [writerWP_bind, writerWP_read]
  rw [writerWP_getNode_known x n c w hlk]
  split
  · simp only [writerWP_bind]; exact writerWP_throw _ _ _ _
  · rename_i hft
    split
    · exact writerWP_throw _ _ _ _
    · rename_i only honly
      split
      · simp only [writerWP_bind]; exact writerWP_throw _ _ _ _
      · rename_i hcv
        simp only [writerWP_bind]
        rw [writerWP_note_iff]
        exact h only (by simpa using hft) honly (by simpa using hcv) _

theorem left_mtCollapse (B : NodeId) (x : NodeId) (d : Nat) (c : Ctx) (w : W) (n : KNode)
    (hlk : w.nodes.lookup x = some n) (hnl : n.leaf = false)
    (hv : ViewGood B c) (hp : MergeParamsOK c.p) (hl : ViewLeftOK c.p c.view)
    (hw : CacheGood B c w) (hb : Book B c w) (hn : LeftCacheE [] w) :
    WriterWP (Mt.collapse x d) c w (fun _ v => CacheGood B c v ∧ Book B c v ∧ Kept w v ∧
      LeftCacheE [] v) := by
  have hpos := hv.pos
  apply mtCollapse_wp x d c w n hlk
  intro only hft _ _ s
  have hmem := bk_mem_of_lookup _ _ _ hlk
  have hbase := left_base_ne B c x n (hw.2 _ hmem) hpos (hn _ hmem) (by rw [hnl]; simp) (by simp [hft])
  have hb' := (book_note B s c w hv.ids hb () { w with trail := s :: w.trail } (run_note s c w)).1
  refine writerWP_mono (left_replaceRoot B x d only c _ hv hp hl hw hb' hn hbase) ?_
  rintro _ v ⟨h1, h2, h3, h4⟩
  exact ⟨h1, h2, (Kept.same rfl rfl).trans h3, h4⟩

theorem sized_mtCollapse (x : NodeId) (d : Nat) : KeepsSized (Mt.collapse x d) := by
  intro c w hv hw
  unfold Mt.collapse
  repeat' first
    | wpSimplify
    | szCore
    | szUse (sized_replaceRoot _ _ _)
    | wpSplit
    | split
    | contradiction

theorem dw_mtCollapse (base : View BVal) (x : NodeId) (d : Nat) : KeepsDW base (Mt.collapse x d) := by
  intro c w hvc hth hw
  unfold Mt.collapse
  repeat' (first | dwStep | dwStep2 | (refine writerWP_mono (dw_replaceRoot base _ _ _ c _ hvc hth (by assumption)) ?_; intro _ _ h; exact h) | split)

theorem kd_mtCollapse (K : KindOracle) (B : NodeId) (x : NodeId) (d : Nat) (c : Ctx) (w : W)
    (n : KNode) (hlk : w.nodes.lookup x = some n) (hnl : n.leaf = false)
    (hv : ViewIdsBelow B c.view) (hk : ViewKinds K B c) (hb : Book B c w) (hi : KdInv K B w)
    (a : Option Nat) (hpx : w.path[d]? = some (a, x)) :
    WriterWP (Mt.collapse x d) c w (fun _ v => Book B c v ∧ KdInv K B v) := by
  apply mtCollapse_wp x d c w n hlk
  intro only _ _ _ s
  obtain ⟨g, h, hm⟩ := kd_tag_of_lookup hb hlk
  have hh : h ≠ 0 := by
    intro h0; subst h0
    have := kd_leaf hb hlk hm
    rw [hnl] at this; simp at this
  obtain ⟨hb', hi', _⟩ := kd_note K B s c w hv hk hb hi () { w with trail := s :: w.trail } (run_note s c w)
  refine writerWP_mono (kd_replaceRoot K B x d only c _ hv hk hb' hi' a hpx g h hm hh) ?_
  rintro _ v ⟨h1, h2, _⟩
  exact ⟨h1, h2⟩

theorem ht_mtCollapse {H : NodeId → Nat → Prop} (B : NodeId) (x : NodeId) (d : Nat) (c : Ctx) (w : W)
    (n : KNode) (hlk : w.nodes.lookup x = some n) (hnl : n.leaf = false)
    (hvo : ViewHtOK H B c) (hpos : 0 < c.i) (hb : Book B c w) (hi : HtInv H B c w)
    (a : Option Nat) (hpx : w.path[d]? = some (a, x)) (hpf : PathFreeTo w (d + 1))
    (hseg : ∀ a' q, d ≠ 0 → w.path[d - 1]? = some (a', q) → ∃ g', (q, g', 0) ∈ w.gh) :
    WriterWP (Mt.collapse x d) c w (fun _ v => Book B c v ∧ HtInv H B c v) := by
  apply mtCollapse_wp x d c w n hlk
  intro only _ honly hcv s
  obtain ⟨g, h, hm⟩ := kd_tag_of_lookup hb hlk
  obtain ⟨hb', hi', _, _, _⟩ := ht_note B s c w hvo hb hi () { w with trail := s :: w.trail } (run_note s c w)
  refine writerWP_mono (ht_replaceRoot B x d only c _ hvo hpos hb' hi' g h hm n hlk hnl honly hcv a hpx
    hpf hseg) ?_
  rintro _ v ⟨h1, h2, _⟩
  exact ⟨h1, h2⟩

/-- A fix step that made a collapse: the path, the target, the collapse. -/
theorem fixStep_collapse_split (k : Key) (c : Ctx) (w0 w : W) (x : NodeId) (d : Nat) (w1 : W)
    (hfs : fixStep k c w0 = .ok (true, w))
    (hpt : (do path k; target : M Fix) c w0 = .ok (.collapse x d, w1)) :
    ∃ u w2, path k c w0 = .ok (u, w2) ∧ target c w2 = .ok (.collapse x d, w1) ∧
      Mt.collapse x d c w1 = .ok ((), w) := by
  obtain ⟨u, w2, hp, ht⟩ := (m_bind_ok _ _ _ _ _).mp hpt
  unfold fixStep at hfs
  obtain ⟨u', w2', hp', hrest⟩ := (m_bind_ok _ _ _ _ _).mp hfs
  rw [hp] at hp'
  simp only [Except.ok.injEq, Prod.mk.injEq] at hp'
  obtain ⟨_, rfl⟩ := hp'
  obtain ⟨f', w1', ht', hm⟩ := (m_bind_ok _ _ _ _ _).mp hrest
  rw [ht] at ht'
  simp only [Except.ok.injEq, Prod.mk.injEq] at ht'
  obtain ⟨rfl, rfl⟩ := ht'
  obtain ⟨m, w3, hmm, hpure⟩ := (m_bind_ok _ _ _ _ _).mp hm
  simp only [pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq, Prod.mk.injEq] at hpure
  obtain ⟨_, rfl⟩ := hpure
  exact ⟨u, w2, hp, ht, hmm⟩

theorem target_collapse (c : Ctx) (base : View BVal) (hvc : ViewCoherent c base) (w : W)
    (ht : ∀ y, w.taken.contains y = false) (x : NodeId) (d : Nat) (w1 : W)
    (h : target c w = .ok (.collapse x d, w1)) :
    (∃ a xn, w.path[d]? = some (a, x) ∧ wNode c.p base w x = some xn ∧ xn.leaf = false) ∧
      (d = 0 ∨ ∃ q qn, w.path[d - 1]? = some q ∧ wNode c.p base w q.2 = some qn ∧ qn.leaf = true) := by
  unfold target at h
  obtain ⟨n, w3, hsz, hgo⟩ := (m_bind_ok _ _ _ _ _).mp h
  simp only [pathSize, bind, ReaderT.bind, StateT.bind, get, getThe, MonadStateOf.get,
    liftM, monadLift, MonadLift.monadLift, StateT.get, pure, ReaderT.pure, StateT.pure,
    Except.pure, Except.bind] at hsz
  have hsz' : (Except.ok (w.path.length, w) : Except String (Nat × W)) = .ok (n, w3) := hsz
  simp only [Except.ok.injEq, Prod.mk.injEq] at hsz'
  obtain ⟨rfl, rfl⟩ := hsz'
  exact targetGo_collapse c base hvc _ w ht _ w1 hgo x d rfl

/-! ## The lift and the parts of `Sub` after a collapse step -/

theorem ne_fix_collapse (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (k : Key)
    (z : NodeId) (top : Top) (w : W) (x : NodeId) (d : Nat) (w1 : W) (hfresh : ∀ q ∈ P, q.1 < base)
    (htimes : StructTimesLE P t) (hL : RunLinksOK p P t)
    (htop : topOf p (ctxAt p P t).view z = .ok top)
    (hfs : fixStep k (ctxAt p P t) (wStart base top) = .ok (true, w))
    (hpt : (do path k; target : M Fix) (ctxAt p P t) (wStart base top) = .ok (.collapse x d, w1)) :
    ∀ q ∈ w.nodes, q.2.ks ≠ [] := by
  have hB := putBase_at p P t base hfresh htimes hL
  have hb0 := putBook_at p P t base hfresh htimes hL z top htop
  obtain ⟨u, w2, hp, ht, hc⟩ := fixStep_collapse_split k _ _ w x d w1 hfs hpt
  have h1 := nex_path base k [] _ _ hB.good hb0.good (by intro q hq; cases hq) u w2 hp
  have h2 := nex_target base [] _ w2 hB.good h1.1 h1.2 _ w1 ht
  have h3 := nex_mtCollapse base x d [] _ w1 hB.good h2.1 h2.2 () w hc
  intro q hq hk
  have := h3.2 q hq hk
  cases this

/-- The lift of a fix step that made a collapse. -/
theorem t_liftFix_collapse (p : Params) : LiftFix p IsCollapse := by
  intro s k base w f w1 hs hI hfresh hrun hfix hC
  obtain ⟨hsub, htimes, hK, hD, hEnc, _⟩ := hI
  obtain ⟨z, top, hz, htop, hfs, hout, hfo⟩ := runFix_true p s.P s.t base k w hrun
  obtain ⟨x, d, rfl⟩ := hC
  have hpt := hfo _ w1 hfix
  have hroot0 := rootAt_of_top p s.P s.t htimes z top hz htop
  have hne := ne_fix_collapse p s.P s.t base k z top w x d w1 hfresh htimes hsub.links htop hfs hpt
  exact liftOK_of_end p s.P s.t base top w hfresh htimes hsub.links hD hK hEnc (reached_root p s hs)
    hroot0 (endW_fix p s.P s.t base k z top w hfresh htimes hsub.links hEnc htop hfs hout hne)

/-- B3's parts for the collapse. -/
theorem t_partsFix_collapse (p : Params) (hp : MergeParamsOK p) (hth : 2 ≤ p.theta ∨ p.theta = 0) :
    PartsFix p IsCollapse := by
  intro s k rest base w f w1 hs hI hpend hfresh hrun hfix hC
  have hlift := t_liftFix_collapse p s k base w f w1 hs hI hfresh hrun hfix hC
  have hkey : k ≠ [] := key_ne_nil_of_wf k (reached_pend_wf p s hs k (by rw [hpend]; exact List.mem_cons_self))
  obtain ⟨hsub, htimes, _, _, hEnc, _⟩ := hI
  rw [hpend] at hsub
  obtain ⟨z, top, hz, htop, hfs, hout, hfo⟩ := runFix_true p s.P s.t base k w hrun
  obtain ⟨x, d, rfl⟩ := hC
  have hpt := hfo _ w1 hfix
  have hrootL : ∃ r, s.P.get? root = some r ∧ p.keyMax < r.level := by
    obtain ⟨nd, h1, h2, _⟩ := reached_root p s hs
    exact ⟨nd, h1, by rw [h2]; exact rootLevel_gt p⟩
  have hvc := viewCoherent_at p s.P s.t htimes
  have hN : wNode p (viewOf s.P) (wStart base top) = knodeOf p (viewOf s.P) := wNode_wStart _ _ _ _
  have hwf0 : ∀ y yn, wNode p (viewOf s.P) (wStart base top) y = some yn → klongsWf yn.ks = true := by
    intro y yn hy
    rw [hN] at hy
    obtain ⟨nd, hg, hd, hlv, hn⟩ := knodeOf_viewOf p _ y yn hy
    subst hn
    exact (hsub.links.2.1 y nd hg hd hlv).1
  obtain ⟨hr1, _, _, hcached, _, _⟩ := writerWP_pathTarget k hkey (ctxAt p s.P s.t) (viewOf s.P) hvc
    (wStart base top) (fun _ _ => rfl) (fun y yn _ hy => hwf0 y yn hy) _ w1 hpt
  obtain ⟨u, w2, hpth, htg, hcl⟩ := fixStep_collapse_split k _ _ w x d w1 hfs hpt
  obtain ⟨hr2, _, _⟩ := path_path (ctxAt p s.P s.t) (viewOf s.P) hvc k (wStart base top)
    (fun _ => rfl) hwf0 u w2 hpth
  have ht2 : ∀ y, w2.taken.contains y = false := by intro y; rw [hr2.2.2.1]; rfl
  obtain ⟨⟨a, xn, hxp, hxn, hxl⟩, hseg0⟩ := target_collapse _ _ hvc w2 ht2 x d w1 htg
  have hB := putBase_at p s.P s.t base hfresh htimes hsub.links
  have hb0 := putBook_at p s.P s.t base hfresh htimes hsub.links z top htop
  have hv : ViewIdsBelow base (ctxAt p s.P s.t).view :=
    viewIdsBelow_rootless base _ (viewIdsBelow_view p s.P s.t base hfresh)
  have hk := viewKinds_of p s.P s.t base hfresh htimes hsub.links.1 hsub.kinds
  obtain ⟨hkb2, hki2, _⟩ := kd_path (kindOracleOf p s.P) base k _ _ hv hk (book_start base _ top)
    (kdInv_start _ base top) u w2 hpth
  obtain ⟨hkb1, hki1, _, hp12⟩ := kd_target (kindOracleOf p s.P) base _ w2 hv hk hkb2 hki2 _ w1 htg
  have hxp1 : w1.path[d]? = some (a, x) := by rw [hp12]; exact hxp
  have hw12 : wNode p (viewOf s.P) w1 = wNode p (viewOf s.P) w2 := hr1.1.trans hr2.1.symm
  have ht1 : ∀ y, w1.taken.contains y = false := by intro y; rw [hr1.2.2.1]; rfl
  -- a cached node of the path reads as the writer's node
  have hcache : ∀ e, e ∈ w1.path → ∀ m, wNode p (viewOf s.P) w2 e.2 = some m →
      w1.nodes.lookup e.2 = some m := by
    intro e he m hm
    obtain ⟨n1, hn1⟩ := hcached e he
    have h1 : wNode p (viewOf s.P) w1 e.2 = some n1 := by
      simp only [wNode, ht1 e.2, Bool.false_eq_true, ↓reduceIte, hn1]
    rw [hw12, hm] at h1
    rw [hn1, h1]
  have hxlk : w1.nodes.lookup x = some xn :=
    hcache (a, x) (List.mem_of_getElem? hxp1) xn hxn
  have hE := endW_fix p s.P s.t base k z top w hfresh htimes hsub.links hEnc htop hfs hout
    (ne_fix_collapse p s.P s.t base k z top w x d w1 hfresh htimes hsub.links htop hfs hpt)
  -- the kinds
  have hKW := (kd_mtCollapse (kindOracleOf p s.P) base x d _ w1 xn hxlk hxl hv hk hkb1 hki1 a hxp1
    () w hcl).2.cache
  -- the heights
  have hvo := viewHtOK_at p s.P s.t base hfresh htimes hsub.links hsub.kinds hsub.heights
  obtain ⟨hhb2, hhi2, _⟩ := ht_path (H := htOracleOf p s.P) base k _ _ hvo (book_start base _ top)
    (htInv_start _ base _ top) hB.wf (by intro q hq; simp [wStart] at hq) u w2 hpth
  obtain ⟨hhb1, hhi1, _, _⟩ := ht_target base _ w2 hvo hhb2 hhi2 _ w1 htg
  have hseg : ∀ a' q, d ≠ 0 → w1.path[d - 1]? = some (a', q) → ∃ g', (q, g', 0) ∈ w1.gh := by
    intro a' q hd hq
    rcases hseg0 with h0 | ⟨q0, qn, hq0, hqn, hql⟩
    · exact absurd h0 hd
    · rw [hp12, hq0] at hq
      simp only [Option.some.injEq] at hq
      subst hq
      have hqlk := hcache _ (by rw [hp12]; exact List.mem_of_getElem? hq0) qn hqn
      obtain ⟨g', h', hm⟩ := kd_tag_of_lookup hhb1 hqlk
      have := kd_leaf hhb1 hqlk hm
      rw [hql] at this
      have h0 : h' = 0 := by simpa using this.symm
      subst h0
      exact ⟨g', hm⟩
  have hHt := (ht_mtCollapse base x d _ w1 xn hxlk hxl hvo (by show 0 < s.t + 1; omega) hhb1 hhi1 a
    hxp1 (by intro _ _ y _ _; exact ht1 y) hseg () w hcl).2.cache
  -- the sizes
  have hvs : ViewSized (ctxAt p s.P s.t) :=
    ⟨hp, viewSizeOK_rootless p _ (sizeStep_view p s.P s.t htimes hsub.size)⟩
  have hs2 := sized_path k _ _ hvs (by intro q hq; simp [wStart] at hq) u w2 hpth

    intro hθ y yn hy

  refine ⟨kindsAllOK_e p s.P (s.t + 1) base top w (by omega) hfresh htimes hrootL hsub.links.2.1

    heightAllOK_e p s.P (s.t + 1) base top w (by omega) hfresh htimes hrootL hsub.links.2.1
