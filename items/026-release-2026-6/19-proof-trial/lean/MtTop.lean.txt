theorem refGet_init (p : Params) (t : Time) (key : Key) : refGet (St.init p).ref t key = none := by

/-- The first state has the run's invariant. -/
theorem stInv_init (p : Params) (hM : 2 ≤ p.M) : StInv p (St.init p) := by
  have htimes : StructTimesLE (initStruct p) 0 := treeRun_timesLE p []
  refine ⟨?_, htimes, treeRun_highKindsOK p [], treeRun_dataEntriesOK p [],
    structEncoded_viewOf _ (treeRun_encoded p [] (fun tx ht => by cases ht)), ?_⟩
  · exact t_sub_of_full p (initStruct p) 0 htimes
      ⟨runLinksOK_init p, treeAt_init p, latestQD_init p hM⟩
  · intro key _
    show get p (viewOf (initStruct p)) 0 key = refGet (St.init p).ref 1 key
    rw [get_init, refGet_init p]

/-! ## A fix step's kind -/

/-- A fix step that made a fix chose one of the three kinds. -/
theorem fix_kind (p : Params) (base : NodeId) (k : Key) (i : Time) (v : View BVal) (w : W)
    (h : runFix p base k i v = .ok (true, w)) :
    ∃ f w1, fixOf p base k i v = .ok (f, w1) ∧ (IsMerge f ∨ IsUnlink f ∨ IsCollapse f) := by
  unfold runFix at h
  split at h
  · cases h
  unfold runWith at h
  unfold fixOf runWith
  split at h
  · cases h
  rename_i z _
  split at h
  · cases h
  rename_i top _
  simp only [ReaderT.run, StateT.run] at h ⊢
  obtain ⟨a, w', hfs, hb⟩ := (m_bind_ok _ _ _ _ _).mp h
  obtain ⟨u, w'', _, hp⟩ := (m_bind_ok _ _ _ _ _).mp hb
  have ha : a = true := by
    simp only [pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq, Prod.mk.injEq] at hp
    exact hp.1
  subst ha
  unfold fixStep at hfs
  obtain ⟨u1, w1, hpath, hrest⟩ := (m_bind_ok _ _ _ _ _).mp hfs
  obtain ⟨f, w2, htgt, hmatch⟩ := (m_bind_ok _ _ _ _ _).mp hrest
  refine ⟨f, w2, (m_bind_ok _ _ _ _ _).mpr ⟨u1, w1, hpath, htgt⟩, ?_⟩
  cases f with
  | none =>
    simp only [pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq, Prod.mk.injEq] at hmatch
    cases hmatch.1
  | merge x d => exact Or.inl ⟨x, d, rfl⟩
  | unlink d => exact Or.inr (Or.inl ⟨d, rfl⟩)
  | collapse x d => exact Or.inr (Or.inr ⟨x, d, rfl⟩)

/-! ## The induction and the top -/

/-- The induction: every reached state has the run's invariant. -/
theorem t_reached_inv (p : Params) : T_reached_inv p := by
  intro hp _ hU hM hUn hC hD s hs
  induction hs with
  | init => exact stInv_init p (by have := hp.2.2.2.1; omega)
  | step s s' hs hstep ih =>
    cases hstep with
    | user tx noops w h1 h2 h3 h4 h5 => exact hU s tx noops w hs ih h1 h2 h3 h4 h5
    | fix k rest base w h1 h2 h3 =>
      obtain ⟨f, w1, hf, hk⟩ := fix_kind p base k (s.t + 1) _ w h3
      rcases hk with hk | hk | hk
      · exact hM s k rest base w f w1 hs ih h1 h2 h3 hf hk
      · exact hUn s k rest base w f w1 hs ih h1 h2 h3 hf hk
      · exact hC s k rest base w f w1 hs ih h1 h2 h3 hf hk
    | done k rest base w h1 h2 h3 => exact hD s k rest base w hs ih h1 h2 h3

/-- The trial's top statement, reduced to the obligations of its premises. -/
theorem t_visible (p : Params) : T_visible p := by
  intro hp hth hU hM hUn hC hD s hs hpend
  have hI := t_reached_inv p hp hth hU hM hUn hC hD s hs
  have hsub := hI.1
  rw [hpend] at hsub
  exact ⟨t_full_of_sub p s.P s.t hI.2.1 hsub, hI.2.2.2.2.2, t_scans_of_full p s hs hI hpend⟩

/-- The top with the merge's step at the writer's state (`FixW`, prover A's A3 and A4;

    MergeParamsOK p → (2 ≤ p.theta ∨ p.theta = 0) →
