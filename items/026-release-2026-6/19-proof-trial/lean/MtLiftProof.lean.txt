/-
The proof trial of 026-18 (bergdb 026-19, step B1): the lift of a subtransaction's last
state to the structure, for any structure P at t and any writer program whose last state
has the books that `execW_*` gives today's transactions (`EndW`); and the books of MT's
fix step that made a merge (MtBooks.lean). Today's proofs, stated for `treeRun` and
`execW`, are restated here with the books as hypotheses ("restated from NAME").
-/

/-- The writer's context of a subtransaction at time i from P at i - 1 (`ctxAt`). -/
def cx (p : Params) (P : Struct BVal) (i : Time) : Ctx :=
  { p := p, i := i, view := rootless (view (treeSpec p) P (i - 1)) }

/-- The books of a subtransaction's last state that the lift needs: what `execW_good`,
`execW_klongsWf`, `execW_allocatedIds`, `execW_ledger`, `execW_book`, `madeNodes_of_runOps`,
`commitInv_of_runOps`, `execW_madeData`, `execW_madeKinds`, `execW_ptrsOK`, `execW_nonempty`
and the encoding give today; nothing taken back and nothing outside the level bounds. -/
structure EndW (p : Params) (P : Struct BVal) (i : Time) (base : NodeId) (top : Top) (w : W) : Prop where
  good : CacheGood base (cx p P i) w
  kwf : WriterKlongsWf w
  alloc : AllocatedIds base w
  ledger : Ledger (cx p P i) w
  book : Book base (cx p P i) w
  taken0 : w.taken = []
  out0 : w.outside = []
  made : MadeNodes (cx p P i) w
  commit : CommitInv top (cx p P i) w
  data : MadeData w
  kinds : MadeKinds (cx p P i) w
  ptrs : PtrsOK (fun y => (P.get? y).isSome = true) (cx p P i) w
  ne : ∀ q ∈ w.nodes, q.2.ks ≠ []
  enc : WriterEncoded w

theorem EndW.nodup {p : Params} {P : Struct BVal} {i : Time} {base : NodeId} {top : Top} {w : W}
    (h : EndW p P i base top w) : (w.nodes.map (·.1)).Nodup := by
  rw [h.book.ids]
  exact ht_nodup_reverse h.book.nodup

theorem EndW.takenAbove {p : Params} {P : Struct BVal} {i : Time} {base : NodeId} {top : Top} {w : W}
    (h : EndW p P i base top w) : ∀ x ∈ w.taken, base ≤ x := by
  rw [h.taken0]; intro x hx; cases hx

theorem EndW.notTaken {p : Params} {P : Struct BVal} {i : Time} {base : NodeId} {top : Top} {w : W}
    (h : EndW p P i base top w) (x : NodeId) : w.taken.contains x = false := by
  rw [h.taken0]; rfl

theorem EndW.outT {p : Params} {P : Struct BVal} {i : Time} {base : NodeId} {top : Top} {w : W}
    (h : EndW p P i base top w) : ∀ q ∈ w.outside, w.taken.contains q.1 = true := by
  rw [h.out0]; intro q hq; cases hq

-- restated from apply_keyNode (ApplyBridge.lean)
theorem apply_keyNode_e (p : Params) (P : Struct BVal) (i : Time) (base : NodeId) (top : Top) (w : W)
    (hpos : 0 < i) (hfresh : ∀ q ∈ P, q.1 < base) (htimes : StructTimesLE P (i - 1))
    (hroot : ∃ r, P.get? root = some r ∧ p.keyMax < r.level) (hH : KeyHistOK p P)
    (hE : EndW p P i base top w)
    (x : NodeId) (nd' : Node BVal) (hx : (P.apply i (changesOf i w)).get? x = some nd')
    (hd : nd'.isData = false) (hlv : nd'.level ≤ p.keyMax) :
    (∃ n, w.nodes.lookup x = some n ∧ w.taken.contains x = false ∧
      nd'.hist = entriesOf n.ks ∧ klongsOf nd'.hist = n.ks ∧ klongsWf n.ks = true) ∨
    (P.get? x = some nd' ∧ (w.nodes.lookup x = none ∨ w.taken.contains x = true)) := by
  have hgood := hE.good
  have hnodup : (w.nodes.map (·.1)).Nodup := hE.nodup
  have hwf := hE.kwf
  have halloc := hE.alloc
  have hledger := hE.ledger
  obtain ⟨r, hr, hrl⟩ := hroot
  have hrootB : root < base := hfresh _ (mem_of_get? hr)
  rw [get?_apply] at hx
  -- The node before the entries, and what the transaction adds to it.
  have hadds : ∀ nd0, (P.get? x).or (newNodeOf (changesOf i w) x) = some nd0 →
      nd0.isData = false → nd0.level ≤ p.keyMax →
      ((P.get? x = some nd0 ∧ x < base) ∨
        (P.get? x = none ∧ base ≤ x ∧ nd0.hist = [] ∧ w.taken.contains x = false ∧
          ∃ n, (x, n) ∈ w.nodes)) ∧
      addsFor (changesOf i w) i x =
        match w.nodes.lookup x with
        | some n =>
          if w.taken.contains x then [] else (entriesOf n.ks).filter (fun e => e.time == i)
        | none => [] := by
    intro nd0 h0 hd0 hl0
    cases hpx : P.get? x with
    | some old =>
      rw [hpx] at h0
      have hold : old = nd0 := by simpa using h0
      subst hold
      have hxB : x < base := hfresh _ (mem_of_get? hpx)
      refine ⟨Or.inl ⟨rfl, hxB⟩, ?_⟩
      apply addsFor_changesOf i w x hnodup
      · intro q hq he'
        have := (halloc.2.2 _ (hledger.2.1 q hq)).1
        simp only at this
        omega
      · intro hnt
        obtain ⟨lvl, _, hm⟩ := hledger.2.2 hnt
        have := (halloc.2.2 _ hm).1
        refine ⟨?_, by simp only at this; omega⟩
        intro hxr
        rw [hxr, hr] at hpx
        have : r = old := by simpa using hpx
        subst this
        exact absurd hl0 (by omega)
    | none =>
      rw [hpx] at h0
      have h0' : newNodeOf (changesOf i w) x = some nd0 := by simpa using h0
      obtain ⟨nn, hnn, hid, hnd0⟩ := newNodeOf_new_mem _ x nd0 h0'
      obtain ⟨hmade, hnt⟩ := changesOf_new_mem i w nn hnn
      subst hid
      have hB := (halloc.2.2 nn hmade).1
      have hnnd : nn.isData = false := by rw [hnd0] at hd0; exact hd0
      have hnnl : nn.level ≤ p.keyMax := by rw [hnd0] at hl0; exact hl0
      obtain ⟨n0, hn0⟩ := hledger.1 nn hmade hnnd hnnl
      refine ⟨Or.inr ⟨rfl, hB, by rw [hnd0], hnt, n0, hn0⟩, ?_⟩
      apply addsFor_changesOf i w nn.id hnodup
      · intro q hq he'
        have hq' := hledger.2.1 q hq
        have := made_unique base w halloc _ _ hq' hmade he'
        rw [← this] at hnnd
        cases hnnd
      · intro hnt'
        obtain ⟨lvl, hlvl, hm⟩ := hledger.2.2 hnt'
        refine ⟨by omega, ?_⟩
        intro hxt
        have := made_unique base w halloc _ _ hm hmade hxt.symm
        rw [← this] at hnnl
        have hlvl' : p.keyMax < lvl := hlvl
        simp only at hnnl
        omega
  cases h0 : (P.get? x).or (newNodeOf (changesOf i w) x) with
  | none => simp [h0] at hx
  | some nd0 =>
    rw [h0] at hx
    have hnd' : nd' = { nd0 with hist := nd0.hist ++ addsFor (changesOf i w) i x } := by
      simpa using hx.symm
    have hd0 : nd0.isData = false := by rw [hnd'] at hd; exact hd
    have hl0 : nd0.level ≤ p.keyMax := by rw [hnd'] at hlv; exact hlv
    obtain ⟨hcase, hadd⟩ := hadds nd0 h0 hd0 hl0
    cases hlook : w.nodes.lookup x with
    | none =>
      rw [hlook] at hadd
      rcases hcase with ⟨hp0, _⟩ | ⟨_, _, _, _, n, hn⟩
      · refine Or.inr ⟨?_, Or.inl rfl⟩
        rw [hp0, hnd', hadd, List.append_nil]
      · have := List.lookup_eq_none_iff.mp hlook (x, n) hn
        simp at this
    | some n =>
      rw [hlook] at hadd
      cases htk : w.taken.contains x with
      | true =>
        simp only [htk, ↓reduceIte] at hadd
        rcases hcase with ⟨hp0, _⟩ | ⟨_, _, _, hnt, _⟩
        · refine Or.inr ⟨?_, Or.inr rfl⟩
          rw [hp0, hnd', hadd, List.append_nil]
        · rw [htk] at hnt
          cases hnt
      | false =>
        simp only [htk, Bool.false_eq_true, ↓reduceIte] at hadd
        have hg := cacheGood_lookup base _ w hgood x n hlook
        have hwn := writerKlongsWf_lookup w hwf x n hlook
        obtain ⟨hsplit, _⟩ := entriesOf_split n.ks i hpos hwn hg.times
        have hold := hg.old
        simp only [cx] at hold
        have hbase : entriesOf (baseKlongs (rootless (view (treeSpec p) P (i - 1))) x) = nd0.hist := by
          rcases hcase with ⟨hp0, hxB⟩ | ⟨hp0, hxB, hh0, _, _⟩
          · rcases hg.origin with ⟨ndv, hv, _, _⟩ | hge
            · have hv' := view_eq_get? p P (i - 1) htimes x ndv (rootless_some_eq _ x ndv hv)
              rw [hp0] at hv'
              have : nd0 = ndv := by simpa using hv'
              subst this
              have hb : baseKlongs (rootless (view (treeSpec p) P (i - 1))) x = klongsOf nd0.hist := by
                have hv2 : rootless (view (treeSpec p) P (i - 1)) x = some nd0 := hv
                unfold baseKlongs
                rw [hv2]
              rw [hb]
              exact (hH x nd0 hp0 hd0 hl0).2
            · omega
          · have hnone : rootless (view (treeSpec p) P (i - 1)) x = none := by
              cases hv : rootless (view (treeSpec p) P (i - 1)) x with
              | none => rfl
              | some ndv =>
                have hv' := view_eq_get? p P (i - 1) htimes x ndv (rootless_some_eq _ x ndv hv)
                rw [hp0] at hv'
                cases hv'
            have hb : baseKlongs (rootless (view (treeSpec p) P (i - 1))) x = [] := by
              unfold baseKlongs
              rw [hnone]
            rw [hb, hh0]
            rfl
        have hhist : nd'.hist = entriesOf n.ks := by
          rw [hnd', hadd]
          show nd0.hist ++ _ = _
          rw [← hbase, ← hold]
          exact hsplit.symm
        refine Or.inl ⟨n, rfl, rfl, hhist, ?_, hwn⟩
        rw [hhist]
        exact klongsRoundTrip n.ks hwn

-- restated from overlayLift_of_book (OverlayLift.lean)
theorem overlayLift_e (p : Params) (P : Struct BVal) (i : Time) (base : NodeId) (top : Top)
    (w : W) (hpos : 0 < i) (hfresh : ∀ q ∈ P, q.1 < base) (htimes : StructTimesLE P (i - 1))
    (hroot : ∃ r, P.get? root = some r ∧ p.keyMax < r.level) (hH : KeyHistOK p P)
    (hE : EndW p P i base top w) :
    (∀ x n, w.nodes.lookup x = some n → w.taken.contains x = false →
      knodeOf p (viewOf (P.apply i (changesOf i w))) x = some n) ∧
    (∀ x ∈ w.taken, base ≤ x) ∧
    (w.newTop = true → w.taken.contains w.top.id = false) := by
  have hbook := hE.book
  refine ⟨?_, hE.takenAbove, ?_⟩
  · intro x n hlook hnt
    have hmem := mem_ids_of_lookup w.nodes x n hlook
    have hxg : x ∈ (w.gh.map (·.1)).reverse := by
      rw [← hbook.ids]
      exact List.mem_map.mpr ⟨(x, n), hmem, rfl⟩
    obtain ⟨⟨x', g, h⟩, hgh, hx'⟩ := List.mem_map.mp (List.mem_reverse.mp hxg)
    simp only at hx'
    subst hx'
    have hleaf := hbook.leaf x' n g h hmem hgh
    -- The node of the structure after the transaction, of a key level.
    have hnode : ∃ nd', (P.apply i (changesOf i w)).get? x' = some nd' ∧ nd'.isData = false ∧
        nd'.level ≤ p.keyMax ∧ p.heightOf nd'.level = h := by
      rcases hbook.tag x' g h hgh with ⟨nd, hv, hd, hlv, _, hh⟩ | ⟨hB, hmade, hout⟩
      · have hg := view_eq_get? p P (i - 1) htimes x' nd (rootless_some_eq _ x' nd hv)
        exact ⟨_, get?_apply_of_get? hg, hd, hlv, hh.symm⟩
      · have hin : ¬ (p.G < g ∨ p.H < h) := by
          intro hc
          have := hE.outT _ (hout hc)
          simp only at this
          rw [hnt] at this
          cases this
        have hhH : h ≤ p.H := by omega
        have halloc := hE.alloc
        have hnew := newNodeOf_changesOf_made base i w halloc _ hmade hnt
        have hPx := get?_none_of_ge P base x' hfresh hB
        have hnew' : newNodeOf (changesOf i w) x' =
            some { level := p.keyLevel g h, isData := false, hist := [] } := hnew
        have hget : (P.apply i (changesOf i w)).get? x' = some (Node.mk (p.keyLevel g h) false
            ([] ++ addsFor (changesOf i w) i x')) := by
          rw [get?_apply, hPx, Option.none_or, hnew']
          rfl
        exact ⟨_, hget, rfl, keyLevel_le_keyMax p g h hhH, Params.heightOf_keyLevel p g h hhH⟩
    obtain ⟨nd', hget, hd, hlv, hh⟩ := hnode
    rcases apply_keyNode_e p P i base top w hpos hfresh htimes hroot hH hE x' nd' hget hd hlv with
      ⟨n1, hl1, _, _, hk, _⟩ | ⟨_, hno | htk⟩
    · rw [hlook] at hl1
      have hn1 : n = n1 := by simpa using hl1
      subst hn1
      have hlv' : ¬ p.keyMax < nd'.level := by omega
      simp only [knodeOf, viewOf, hget, hd, hlv', Bool.false_or, decide_false, Bool.false_eq_true,
        ↓reduceIte, hh, hk, Option.some.injEq]
      cases n
      simp only [KNode.mk.injEq, and_true]
      exact hleaf.symm
    · rw [hlook] at hno
      cases hno
    · rw [hnt] at htk
      cases htk
  · intro hnt
    obtain ⟨_, _, hnot⟩ := hbook.top hnt
    cases hc : w.taken.contains w.top.id with
    | false => rfl
    | true =>
      have hm : w.top.id ∈ w.taken := by simpa using hc
      exact (hnot (hbook.taken _ hm)).elim

-- restated from commitLiftAt_of_links (TakeBackRun.lean)
theorem knode_commit_e (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (top : Top) (w : W)
    (hbase : ∀ q ∈ P, q.1 < base) (htimes : StructTimesLE P t) (hH : KeyHistOK p P)
    (hroot : ∃ nd, P.get? root = some nd ∧ nd.level = p.rootLevel ∧ nd.isData = false)
    (hE : EndW p P (t + 1) base top w) :
    ∀ x, knodeOf p (viewOf (P.apply (t + 1) (changesOf (t + 1) w))) x = wNode p (viewOf P) w x := by
  have hrootL : ∃ r, P.get? root = some r ∧ p.keyMax < r.level := by
    obtain ⟨nd, h1, h2, _⟩ := hroot
    exact ⟨nd, h1, by rw [h2]; exact rootLevel_gt p⟩
  obtain ⟨hov, htaken, _⟩ := overlayLift_e p P (t + 1) base top w (by omega) hbase htimes hrootL hH hE
  have halloc : AllocatedIds base w := hE.alloc
  have hmn := hE.made
  have hinv := hE.commit
  obtain ⟨_, _, hnt, hvm, _, _⟩ := hinv
  intro x
  have hov' := hov x
  generalize hi : t + 1 = i at hov' ⊢
  obtain ⟨ndr, hndr, hlr, hdr⟩ := hroot
  have madeGe : ∀ m ∈ w.made, base ≤ m.id := fun m hm => (halloc.2.2 m hm).1
  by_cases hty : w.taken.contains x = true
  · have hge := htaken x (by simpa using hty)
    have hPx := get?_none_of_ge P base x hbase hge
    have hNx := newNodeOf_changesOf_none i w x (fun m _ hid => by rw [← hid] at hty ⊢; exact hty)
    have hty2 : x ∈ w.taken := by simpa using hty
    simp [knodeOf, viewOf, get?_apply, hPx, hNx, wNode, hty2]
  · have hty' : w.taken.contains x = false := by simpa using hty
    have hty3 : x ∉ w.taken := by simpa using hty
    cases hl : w.nodes.lookup x with
    | some n =>
      rw [hov' n hl hty']
      simp [wNode, hty3, hl]
    | none =>
      have hwn : wNode p (viewOf P) w x = knodeOf p (viewOf P) x := by
        simp [wNode, hty3, hl]
      rw [hwn]
      have hnokey : ∀ q ∈ w.nodes, q.1 ≠ x := by
        intro q hq hqx
        have := List.lookup_eq_none_iff.mp hl q hq
        simp [hqx] at this
      cases hPx : P.get? x with
      | some nd =>
        have hlt : x < base := hbase _ (get?_mem P x nd hPx)
        by_cases hxr : x = root
        · subst hxr
          rw [hndr] at hPx
          cases hPx
          have := rootLevel_gt p
          simp [knodeOf, viewOf, get?_apply_of_get? hndr, hndr, hlr, hdr, this]
        · have hA : addsFor (changesOf i w) i x = [] :=
            List.eq_nil_iff_forall_not_mem.mpr (fun e he => by
              obtain ⟨_, hm⟩ := mem_addsFor.mp he
              rcases mem_changesOf_adds i w x e.val hm with ⟨q, hq, _, hqy⟩ | ⟨h', hh', _⟩ |
                  ⟨hnt', hr⟩
              · exact hnokey q hq hqy
              · have := madeGe _ (hvm _ hh')
                simp only at this
                omega
              · rcases hr with ⟨hxr', _⟩ | ⟨hxt, _⟩
                · exact hxr hxr'
                · obtain ⟨m, hm, hmid, _, _⟩ := hnt hnt'
                  have := madeGe m hm
                  rw [hmid, ← hxt] at this
                  omega)
          simp [knodeOf, viewOf, get?_apply_of_get? hPx, hPx, hA]
      | none =>
        have hk0 : knodeOf p (viewOf P) x = none := by simp [knodeOf, viewOf, hPx]
        rw [hk0]
        cases hNx : newNodeOf (changesOf i w) x with
        | none => simp [knodeOf, viewOf, get?_apply, hPx, hNx]
        | some nd =>
          obtain ⟨m, hm, hid, _, hnd⟩ := newNodeOf_changesOf_some i w x nd hNx
          subst hnd
          cases hmd : m.isData with
          | true => simp [knodeOf, viewOf, get?_apply, hPx, hNx, hmd]
          | false =>
            rcases hmn m hm hmd with hs | hlv
            · rw [hid, hl] at hs
              cases hs
            · have hlv' : p.keyMax < m.level := hlv
              simp [knodeOf, viewOf, get?_apply, hPx, hNx, hmd, hlv']

-- restated from rootAt_commit_of_lift (MergeNodeCountOps.lean)
theorem root_commit_e (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (top : Top) (w : W)
    (hbase : ∀ q ∈ P, q.1 < base) (htimes : StructTimesLE P t)
    (hroot : ∃ nd, P.get? root = some nd ∧ nd.level = p.rootLevel ∧ nd.isData = false)
    (hroot0 : rootAt (viewOf P) t = some top.rootKey)
    (hE : EndW p P (t + 1) base top w)
    (hK : ∀ x, knodeOf p (viewOf (P.apply (t + 1) (changesOf (t + 1) w))) x = wNode p (viewOf P) w x) :
    rootAt (viewOf (P.apply (t + 1) (changesOf (t + 1) w))) (t + 1) = some w.top.rootKey := by
  have htaken := hE.takenAbove
  have htopt : w.newTop = true → w.taken.contains w.top.id = false := fun _ => hE.notTaken _
  have halloc : AllocatedIds base w := hE.alloc
  have hinv := hE.commit
  obtain ⟨_, hnf, hnt, hvm, _, _⟩ := hinv
  generalize hi : t + 1 = i at hK ⊢
  obtain ⟨ndr, hndr, hlr, hdr⟩ := hroot
  have noKey : ∀ x, knodeOf p (viewOf (P.apply i (changesOf i w))) x = none →
      w.taken.contains x = false → ∀ q ∈ w.nodes, q.1 ≠ x := by
    intro x hx ht q hq hqx
    rw [hK x] at hx
    simp only [wNode, ht, Bool.false_eq_true, ↓reduceIte] at hx
    cases hl : w.nodes.lookup x with
    | none =>
      have := List.lookup_eq_none_iff.mp hl q hq
      simp [hqx] at this
    | some n => rw [hl] at hx; cases hx
  have madeGe : ∀ m ∈ w.made, base ≤ m.id := fun m hm => (halloc.2.2 m hm).1
  have hrt : w.taken.contains root = false := by
    cases h : w.taken.contains root with
    | false => rfl
    | true =>
      have := htaken root (by simpa using h)
      have hb0 : 0 < base := hbase _ (get?_mem P root ndr hndr)
      simp only [root] at this
      omega
  have hkr : knodeOf p (viewOf (P.apply i (changesOf i w))) root = none := by
    have := rootLevel_gt p
    simp [knodeOf, viewOf, get?_apply_of_get? hndr, hlr, hdr, this]
  have noKeyRoot := noKey root hkr hrt
  have hradds : ∀ e ∈ addsFor (changesOf i w) i root,
      w.newTop = true ∧ e = ⟨i, .top w.top.id⟩ := by
    intro e he
    obtain ⟨het, hm⟩ := mem_addsFor.mp he
    rcases mem_changesOf_adds i w root e.val hm with ⟨q, hq, _, hqy⟩ | ⟨h', hh', _⟩ |
        ⟨hnt', hr⟩
    · exact absurd hqy (noKeyRoot q hq)
    · have := madeGe _ (hvm _ hh')
      have hb0 : 0 < base := hbase _ (get?_mem P root ndr hndr)
      simp only [root] at this
      omega
    · rcases hr with ⟨_, hval⟩ | ⟨hyt, _⟩
      · refine ⟨hnt', ?_⟩
        cases e
        simp only at het hval
        rw [het, hval]
      · obtain ⟨m, hm, hmid, _, _⟩ := hnt hnt'
        have := madeGe m hm
        have hb0 : 0 < base := hbase _ (get?_mem P root ndr hndr)
        rw [hmid, ← hyt] at this
        simp only [root] at this
        omega
  cases hnew : w.newTop with
  | true =>
    obtain ⟨m, hm, hmid, hmd, hml⟩ := hnt hnew
    have htt := htopt hnew
    have hge : base ≤ w.top.id := hmid ▸ madeGe m hm
    have hPt := get?_none_of_ge P base _ hbase hge
    have hNt := newNodeOf_changesOf_made base i w halloc m hm (by rw [hmid]; exact htt)
    rw [hmid] at hNt
    have hkt : knodeOf p (viewOf (P.apply i (changesOf i w))) w.top.id = none := by
      have hml' : p.keyMax < m.level := hml
      simp [knodeOf, viewOf, get?_apply, hPt, hNt, hml']
    have noKeyTop := noKey _ hkt htt
    have hmems := top_mem_changesOf_adds i w hnew
    unfold rootAt topAt rootKeyOf viewOf
    rw [get?_apply_of_get? hndr]
    simp only
    rw [upTo_append]
    have hA : (addsFor (changesOf i w) i root).upTo i = addsFor (changesOf i w) i root :=
      List.filter_eq_self.mpr (fun e he => by
        rw [(mem_addsFor.mp he).1]; simp)
    rw [hA, getLast_append_const (ndr.hist.upTo i) (addsFor (changesOf i w) i root)
      ⟨i, .top w.top.id⟩
      (List.ne_nil_of_mem (a := (⟨i, .top w.top.id⟩ : Entry BVal)) (mem_addsFor.mpr ⟨rfl, hmems.1⟩))
      (fun e he => (hradds e he).2)]
    simp only [Option.bind_some]
    rw [get?_apply, hPt, hNt]
    simp only [Option.none_or, Option.map_some, Option.bind_some, List.nil_append]
    apply findSome_of_all
    · exact ⟨⟨i, .rootKey w.top.rootKey⟩, mem_addsFor.mpr ⟨rfl, hmems.2⟩, rfl⟩
    · intro e he
      obtain ⟨_, hm'⟩ := mem_addsFor.mp he
      rcases mem_changesOf_adds i w w.top.id e.val hm' with ⟨q, hq, _, hqy⟩ | ⟨h', hh', _⟩ |
          ⟨_, hr⟩
      · exact absurd hqy (noKeyTop q hq)
      · have := nodup_map_inj (·.id) w.made halloc.2.1 m _ hm (hvm _ hh') (by simp [hmid])
        rw [this] at hmd
        cases hmd
      · rcases hr with ⟨hyr, _⟩ | ⟨_, hv | ⟨j, z', hv⟩⟩
        · have hb0 : 0 < base := hbase _ (get?_mem P root ndr hndr)
          simp only [root] at hyr; omega
        · right; rw [hv]
        · left; rw [hv]
  | false =>
    have htw := hnf hnew
    have hA : addsFor (changesOf i w) i root = [] :=
      List.eq_nil_iff_forall_not_mem.mpr (fun e he => by
        have := (hradds e he).1
        rw [hnew] at this
        cases this)
    have hshift := topAt_succ_of_timesLE P t htimes
    rw [htw]
    unfold rootAt at hroot0 ⊢
    have htopEq : topAt (viewOf (P.apply i (changesOf i w))) i = topAt (viewOf P) t := by
      rw [← hshift, hi]
      unfold topAt viewOf
      rw [get?_apply_of_get? hndr, hA, List.append_nil, hndr]
    rw [htopEq]
    cases hzt : topAt (viewOf P) t with
    | none => rw [hzt] at hroot0; cases hroot0
    | some z' =>
      rw [hzt] at hroot0
      simp only [Option.bind_some] at hroot0 ⊢
      exact rootKeyOf_apply_some P i (changesOf i w) z' _ hroot0

-- restated from commitGetOK_of_lift (GetCommit.lean)
theorem get_commit_e (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (top : Top) (w : W)
    (hbase : ∀ q ∈ P, q.1 < base) (htimes : StructTimesLE P t) (hD : DataEntriesOK P)
    (hroot : ∃ nd, P.get? root = some nd ∧ nd.level = p.rootLevel ∧ nd.isData = false)
    (hroot0 : rootAt (viewOf P) t = some top.rootKey)
    (hE : EndW p P (t + 1) base top w)
    (hK : ∀ x, knodeOf p (viewOf (P.apply (t + 1) (changesOf (t + 1) w))) x = wNode p (viewOf P) w x)
    (key : Key) :
    get p (viewOf (P.apply (t + 1) (changesOf (t + 1) w))) (t + 1) key = wGet p (t + 1) (viewOf P) w key := by
  have hR0 := root_commit_e p P t base top w hbase htimes hroot hroot0 hE hK
  have htaken := hE.takenAbove
  have halloc : AllocatedIds base w := hE.alloc
  have hdata : MadeData w := hE.data
  have hinv := hE.commit
  obtain ⟨_, hnf, hnt, hvm, hvn, _⟩ := hinv
  generalize hi : t + 1 = i at hK hR0 ⊢
  obtain ⟨ndr, hndr, hlr, hdr⟩ := hroot
  have hbase0 : 0 < base := hbase _ (get?_mem P root ndr hndr)
  -- Key adds only for the writer's own nodes, which the lift reads.
  have noKey : ∀ x, knodeOf p (viewOf (P.apply i (changesOf i w))) x = none →
      w.taken.contains x = false → ∀ q ∈ w.nodes, q.1 ≠ x := by
    intro x hx ht q hq hqx
    rw [hK x] at hx
    simp only [wNode, ht, Bool.false_eq_true, ↓reduceIte] at hx
    cases hl : w.nodes.lookup x with
    | none =>
      have := List.lookup_eq_none_iff.mp hl q hq
      simp [hqx] at this
    | some n => rw [hl] at hx; cases hx
  have madeGe : ∀ m ∈ w.made, base ≤ m.id := fun m hm => (halloc.2.2 m hm).1
  -- The values.
  have hV : valueOf (viewOf (P.apply i (changesOf i w))) = wValue (viewOf P) w := by
    funext y
    by_cases hty : w.taken.contains y = true
    · have hPy := get?_none_of_ge P base y hbase (htaken y (by simpa using hty))
      have hNy := newNodeOf_changesOf_none i w y (fun m _ hid => by rw [← hid] at hty ⊢; exact hty)
      have hty2 : y ∈ w.taken := by simpa using hty
      simp [valueOf, viewOf, get?_apply, hPy, hNy, wValue, hty2]
    · have hty' : w.taken.contains y = false := by simpa using hty
      cases hv : w.vals.lookup y with
      | some h =>
        obtain ⟨l1, l2, hvl, _⟩ := List.lookup_eq_some_iff.mp hv
        have hmem : (y, h) ∈ w.vals := by rw [hvl]; simp
        have hmade := hvm (y, h) hmem
        have hge : base ≤ y := madeGe _ hmade
        have hPy := get?_none_of_ge P base y hbase hge
        have hNy := newNodeOf_changesOf_made base i w halloc _ hmade hty'
        have hk : knodeOf p (viewOf (P.apply i (changesOf i w))) y = none := by
          simp [knodeOf, viewOf, get?_apply, hPy, hNy]
        have hadds : ∀ e ∈ addsFor (changesOf i w) i y, e.val = .data h := by
          intro e he
          obtain ⟨_, hm⟩ := mem_addsFor.mp he
          rcases mem_changesOf_adds i w y e.val hm with ⟨q, hq, _, hqy⟩ | ⟨h', hh', hv'⟩ |
              ⟨hnt', hr⟩
          · exact absurd hqy (noKey y hk hty' q hq)
          · have := nodup_map_inj (·.1) w.vals hvn _ _ hh' hmem rfl
            cases this
            exact hv'
          · rcases hr with ⟨hyr, _⟩ | ⟨hyt, _⟩
            · rw [hyr] at hge; simp only [root] at hge; omega
            · obtain ⟨m, hm, hmid, hmd, _⟩ := hnt hnt'
              have := nodup_map_inj (·.id) w.made halloc.2.1 m _ hm hmade (by simp [hmid, hyt])
              rw [this] at hmd
              cases hmd
        have hex : (⟨i, .data h⟩ : Entry BVal) ∈ addsFor (changesOf i w) i y :=
          mem_addsFor.mpr ⟨rfl, val_mem_changesOf_adds i w y h hmem⟩
        simp only [valueOf, viewOf, get?_apply, hPy, hNy, Option.none_or, Option.map_some,
          ↓reduceIte, List.nil_append, wValue, hty', hv, Bool.false_eq_true]
        apply findSome_of_all
        · exact ⟨_, hex, rfl⟩
        · intro e he
          right
          rw [hadds e he]
      | none =>
        cases hPy : P.get? y with
        | some nd =>
          simp only [valueOf, viewOf, get?_apply_of_get? hPy, wValue, hty', hv, hPy,
            Bool.false_eq_true, ↓reduceIte]
          cases hd : nd.isData with
          | false => rfl
          | true =>
            obtain ⟨e, he, h0, hed⟩ := hD y nd hPy hd
            simp only [↓reduceIte, List.findSome?_append]
            have hs : (nd.hist.findSome? (fun e => match e.val with
                | .data h => some h
                | _ => none)).isSome = true := by
              apply List.findSome?_isSome_iff.mpr
              exact ⟨e, he, by simp [hed]⟩
            cases hf : nd.hist.findSome? (fun e => match e.val with
                | .data h => some h
                | _ => none) with
            | none => rw [hf] at hs; cases hs
            | some h1 => rfl
        | none =>
          cases hNy : newNodeOf (changesOf i w) y with
          | none => simp [valueOf, viewOf, get?_apply, hPy, hNy, wValue, hty', hv]
          | some nd =>
            obtain ⟨m, hm, hid, _, hnd⟩ := newNodeOf_changesOf_some i w y nd hNy
            cases hmd : m.isData with
            | false =>
              simp [valueOf, viewOf, get?_apply, hPy, hNy, hnd, hmd, wValue, hty', hv]
            | true =>
              obtain ⟨h, hh⟩ := hdata m hm hmd
              rw [hid] at hh
              have := List.lookup_eq_none_iff.mp hv _ hh
              simp at this
  have hR : rootAt (viewOf (P.apply i (changesOf i w))) i = some w.top.rootKey := hR0
  -- The get.
  unfold get wGet
  rw [hR]
  simp only [Option.bind_some]
  rw [getFrom_eq_getFromR, hV, show knodeOf p (viewOf (P.apply i (changesOf i w))) =
    wNode p (viewOf P) w from funext hK]

-- restated from pointersOK_step (PointersStep.lean)
theorem pointersOK_e (p : Params) (P : Struct BVal) (i : Time) (base : NodeId) (top : Top) (w : W)
    (hP : PointersOK P) (hE : EndW p P i base top w) :
    PointersOK (P.apply i (changesOf i w)) := by
  have halloc := hE.alloc
  have hledger := hE.ledger
  have hptrs := hE.ptrs
  have hbook := hE.book
  have hT1 : ∀ q ∈ w.nodes, w.taken.contains q.1 = false →
      ∀ cv y, Klong.key cv y ∈ q.2.ks → w.taken.contains y = false :=
    fun _ _ _ _ y _ => hE.notTaken y
  have hT2 : w.taken.contains w.top.rootKey = false := hE.notTaken _
  -- A node of P, or a made node not taken back, is a node after.
  have hold : ∀ y, (P.get? y).isSome = true →
      ((P.apply i (changesOf i w)).get? y).isSome = true := by
    intro y hy
    cases hpy : P.get? y with
    | none => rw [hpy] at hy; cases hy
    | some nd => rw [get?_apply_of_get? hpy]; rfl
  have hmade : ∀ y, y ∈ w.made.map (·.id) → w.taken.contains y = false →
      ((P.apply i (changesOf i w)).get? y).isSome = true := by
    intro y hy ht
    obtain ⟨m, hm, hmy⟩ := List.mem_map.mp hy
    subst hmy
    rw [get?_apply, newNodeOf_changesOf_made base i w halloc m hm ht]
    cases P.get? m.id <;> rfl
  have hboth : ∀ y, ((P.get? y).isSome = true ∨ y ∈ w.made.map (·.id)) →
      w.taken.contains y = false → ((P.apply i (changesOf i w)).get? y).isSome = true := by
    intro y hy ht
    rcases hy with hy | hy
    · exact hold y hy
    · exact hmade y hy ht
  intro q hq e heq y hy
  rcases ptrStep_entry P i (changesOf i w) q hq e heq with ⟨old, hq0, he0⟩ | hadd
  · exact hold y (hP (q.1, old) hq0 e he0 y hy)
  · rw [changesOf_adds] at hadd
    have hadd' := (List.mergeSort_perm _ _).mem_iff.mp hadd
    rcases List.mem_append.mp hadd' with hadd' | htop
    · rcases List.mem_append.mp hadd' with hkey | hval
      · -- A klong of a node of the overlay not taken back.
        unfold keyAdditions at hkey
        obtain ⟨r, hr, hre⟩ := List.mem_flatMap.mp hkey
        obtain ⟨hrn, hrt⟩ := List.mem_filter.mp hr
        have hrt' : w.taken.contains r.1 = false := by simpa using hrt
        obtain ⟨e', he', hee⟩ := List.mem_map.mp hre
        obtain ⟨_, hv⟩ := Prod.mk.inj hee
        have he'' : e' ∈ entriesOf r.2.ks := (List.mem_filter.mp he').1
        rw [← hv] at hy
        obtain ⟨cv, hk⟩ := ptrStep_entriesOf r.2.ks e' he'' y hy
        exact hboth y (hptrs.klongs r hrn cv y hk) (hT1 r hrn hrt' cv y hk)
      · -- A value's data has no pointer.
        unfold valueAdditions at hval
        obtain ⟨r, _, hre⟩ := List.mem_map.mp hval
        obtain ⟨_, hv⟩ := Prod.mk.inj hre
        rw [← hv] at hy
        simp [bptrs] at hy
    · -- The new top node's entries and the root's entry for it.
      unfold topAdditions at htop
      cases hnt : w.newTop with
      | false => simp [hnt] at htop
      | true =>
        simp only [hnt, ↓reduceIte] at htop
        rcases List.mem_cons.mp htop with htop | htop
        · obtain ⟨_, hv⟩ := Prod.mk.inj htop
          rw [hv] at hy
          simp only [bptrs, List.mem_singleton] at hy
          subst hy
          obtain ⟨lvl, _, hm⟩ := hledger.2.2 hnt
          apply hmade _ (List.mem_map.mpr ⟨_, hm, rfl⟩)
          cases htk : w.taken.contains w.top.id with
          | false => rfl
          | true =>
            have hmem : w.top.id ∈ w.taken := by simpa using htk
            exact absurd (hbook.taken _ hmem) (hbook.top hnt).2.2
        rcases List.mem_cons.mp htop with htop | htop
        · obtain ⟨_, hv⟩ := Prod.mk.inj htop
          rw [hv] at hy
          simp only [bptrs, List.mem_singleton] at hy
          subst hy
          exact hboth _ hptrs.rootKey hT2
        · obtain ⟨s, hs, hse⟩ := List.mem_map.mp htop
          obtain ⟨_, hv⟩ := Prod.mk.inj hse
          rw [← hv] at hy
          simp only [bptrs, List.mem_singleton] at hy
          subst hy
          exact hold _ (hptrs.skips s hs)

theorem endW_fix (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (k : Key) (z : NodeId)
    (top : Top) (w : W) (hfresh : ∀ q ∈ P, q.1 < base) (htimes : StructTimesLE P t)

    (htop : topOf p (ctxAt p P t).view z = .ok top)
    (hfs : fixStep k (ctxAt p P t) (wStart base top) = .ok (true, w)) (hout : w.outside = [])
    (hne : ∀ q ∈ w.nodes, q.2.ks ≠ []) :
    EndW p P (t + 1) base top w := by
  have hB := putBase_at p P t base hfresh htimes hL
  have hb0 := putBook_at p P t base hfresh htimes hL z top htop
  have hbk := putBook_of_keeps (fixStep k) (ctxAt p P t) (viewOf P) base top hB (wStart base top) hb0
    (klongsWf_fixStep k) (good_fixStep base k) (book_fixStep base k)
    (@state_fixStep (allocatedIdsInvariant base) k) (@state_fixStep (commitInvariant top) k)
    (freshCache_fixStep k) (ptrs_fixStep _ k) true w hfs
  have hled := @state_fixStep ledgerInvariant k (ctxAt p P t) (wStart base top)
    (show Ledger (ctxAt p P t) (wStart base top) from
      ⟨by intro nn hn; simp [wStart] at hn, by intro q hq; simp [wStart] at hq, by intro h; simp [wStart] at h⟩) true w hfs
  have hmn := @state_fixStep madeNodesInvariant k (ctxAt p P t) (wStart base top)
    (show MadeNodes (ctxAt p P t) (wStart base top) by intro m hm; simp [wStart] at hm) true w hfs
  have hmd := @state_fixStep madeDataInvariant k (ctxAt p P t) (wStart base top)
    (show MadeData (wStart base top) by intro m hm; simp [wStart] at hm) true w hfs
  have hmk := @state_fixStep madeKindsInvariant k (ctxAt p P t) (wStart base top)
    (show MadeKinds (ctxAt p P t) (wStart base top) by intro m hm; simp [wStart] at hm) true w hfs
  have htk := takenP_fixStep (fun l => l = []) k (ctxAt p P t) (wStart base top) rfl true w hfs
  have hen := encoded_fixStep k (ctxAt p P t) (wStart base top) (viewEncoded_rootless p P t htimes hEnc)
    (show WriterEncoded (wStart base top) from
      ⟨by intro q hq; simp [wStart] at hq, by intro q hq; simp [wStart] at hq⟩) true w hfs
  have hpt := ptrs_fixStep (fun y => (P.get? y).isSome = true) k (ctxAt p P t) (wStart base top)
    (viewPtrsOld_at p P t htimes hL.1)
    (ptrs_start _ (ctxAt p P t) base top (topOf_old p P t htimes hL.1 z top htop)) true w hfs
  exact ⟨hbk.good, hbk.klongs, hbk.alloc, hled, hbk.book, htk, hout, hmn, hbk.commit, hmd, hmk,
    hpt.1, hne, hen⟩

/-- A fix step that made a merge: the path, the target, the merge. -/
theorem fixStep_merge_split (k : Key) (c : Ctx) (w0 w : W) (x : NodeId) (d : Nat) (w1 : W)
    (hfs : fixStep k c w0 = .ok (true, w))
    (hpt : (do path k; target : M Fix) c w0 = .ok (.merge x d, w1)) :
    ∃ u w2, path k c w0 = .ok (u, w2) ∧ target c w2 = .ok (.merge x d, w1) ∧
      ∃ m, Mt.merge x d c w1 = .ok (m, w) := by
  obtain ⟨u, w2, hp, ht⟩ := (m_bind_ok _ _ _ _ _).mp hpt
  unfold fixStep at hfs
  obtain ⟨u', w2', hp', hrest⟩ := (m_bind_ok _ _ _ _ _).mp hfs
  rw [hp] at hp'
  simp only [Except.ok.injEq, Prod.mk.injEq] at hp'
  obtain ⟨_, rfl⟩ := hp'
  obtain ⟨f', w1', ht', hm⟩ := (m_bind_ok _ _ _ _ _).mp hrest
  rw [ht] at ht'
  simp only [Except.ok.injEq, Prod.mk.injEq] at ht'
  obtain ⟨rfl, rfl⟩ := ht'
  obtain ⟨m, w3, hmm, hpure⟩ := (m_bind_ok _ _ _ _ _).mp hm
  simp only [pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq, Prod.mk.injEq] at hpure
  obtain ⟨_, rfl⟩ := hpure
  exact ⟨u, w2, hp, ht, m, hmm⟩

/-- No key node of a merge's last state is without klongs (`NEx`, WriterNonempty.lean). -/
theorem ne_fix_merge (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (k : Key) (z : NodeId)
    (top : Top) (w : W) (x : NodeId) (d : Nat) (w1 : W) (hfresh : ∀ q ∈ P, q.1 < base)
    (htimes : StructTimesLE P t) (hL : RunLinksOK p P t)
    (htop : topOf p (ctxAt p P t).view z = .ok top)
    (hfs : fixStep k (ctxAt p P t) (wStart base top) = .ok (true, w))
    (hpt : (do path k; target : M Fix) (ctxAt p P t) (wStart base top) = .ok (.merge x d, w1)) :
    ∀ q ∈ w.nodes, q.2.ks ≠ [] := by
  have hB := putBase_at p P t base hfresh htimes hL
  have hb0 := putBook_at p P t base hfresh htimes hL z top htop
  obtain ⟨u, w2, hp, ht, m, hmm⟩ := fixStep_merge_split k _ _ w x d w1 hfs hpt
  have h1 := nex_path base k [] _ _ hB.good hb0.good (by intro q hq; cases hq) u w2 hp
  have h2 := nex_target base [] _ w2 hB.good h1.1 h1.2 _ w1 ht
  have h3 := nex_mtMerge base x d [] _ w1 hB.good h2.1 h2.2 m w hmm
  intro q hq hk
  have := h3.2 q hq hk
  cases this

/-- Every structure reached has the primordial super root, a node of the root's level. -/
theorem reached_root (p : Params) (s : St) (h : Reached p s) :
    ∃ nd, s.P.get? root = some nd ∧ nd.level = p.rootLevel ∧ nd.isData = false := by
  induction h with
  | init => exact treeRun_root p []
  | step s s' _ hs ih =>
    obtain ⟨nd, h1, h2, h3⟩ := ih
    cases hs with
    | user tx noops w _ _ _ _ _ => exact ⟨_, get?_apply_of_get? h1, h2, h3⟩
    | fix k rest base w _ _ _ => exact ⟨_, get?_apply_of_get? h1, h2, h3⟩
    | done => exact ⟨nd, h1, h2, h3⟩

/-- B1: the lift of a fix step that made a merge. -/
theorem t_liftFix_merge (p : Params) : LiftFix p IsMerge := by
  intro s k base w f w1 hs hI hfresh hrun hfix hC
  obtain ⟨hsub, htimes, hK, hD, hEnc, _⟩ := hI
  obtain ⟨z, top, hz, htop, hfs, hout, hfo⟩ := runFix_true p s.P s.t base k w hrun
  obtain ⟨x, d, rfl⟩ := hC
  have hpt := hfo _ w1 hfix
  have hroot0 := rootAt_of_top p s.P s.t htimes z top hz htop
  have hne := ne_fix_merge p s.P s.t base k z top w x d w1 hfresh htimes hsub.links htop hfs hpt
  exact liftOK_of_end p s.P s.t base top w hfresh htimes hsub.links hD hK hEnc (reached_root p s hs)
    hroot0 (endW_fix p s.P s.t base k z top w hfresh htimes hsub.links hEnc htop hfs hout hne)
