/-! ## The dead klongs after a fix step that made a merge -/

theorem mem_deletedKeys (ops : List Op) (noops : List Bool) (k : Key)
    (h : k ∈ deletedKeys ops noops) : Op.del k ∈ ops := by
  unfold deletedKeys at h
  have h' := (List.mergeSort_perm _ _).mem_iff.mp h
  obtain ⟨⟨op, noop⟩, hm, hf⟩ := List.mem_filterMap.mp h'
  have hop := (List.of_mem_zip hm).1
  cases op with
  | put _ _ => simp at hf
  | del k' =>
    cases noop with
    | true => simp at hf
    | false =>
      simp only [Bool.false_eq_true, ↓reduceIte, Option.some.injEq] at hf
      subst hf
      exact hop

/-- Every pending key of a reached state is a key of a user operation: well formed. -/
theorem reached_pend_wf (p : Params) (s : St) (h : Reached p s) : ∀ k ∈ s.pend, keyWf k = true := by
  induction h with
  | init => intro k hk; cases hk
  | step s s' _ hs ih =>
    cases hs with
    | user tx noops w _ _ hwf _ _ =>
      intro k hk
      have hop := mem_deletedKeys tx.ops noops k hk
      simp only [Tx.wf, List.all_eq_true] at hwf
      exact hwf _ hop
    | fix k rest base w h1 _ _ => intro k' hk'; exact ih k' (by rw [h1]; exact hk')
    | done k rest base w h1 _ _ =>
      intro k' hk'; exact ih k' (by rw [h1]; exact List.mem_cons_of_mem _ hk')

theorem key_ne_nil_of_wf (k : Key) (h : keyWf k = true) : k ≠ [] := by
  intro h0; subst h0; simp [keyWf] at h

/-- The dead klongs after a fix step that made a merge: `dw_mtMerge` from the state after the
path and the target (A's `writerWP_pathTarget`), then the lift. -/
theorem dead_fix_merge (p : Params) (hp : MergeParamsOK p) (hth : 2 ≤ p.theta ∨ p.theta = 0)
    (s : St) (k : Key) (rest : List Key) (base : NodeId) (w : W) (f : Fix) (w1 : W)
    (hs : Reached p s) (hI : StInv p s) (hpend : s.pend = k :: rest) (hfresh : ∀ q ∈ s.P, q.1 < base)
    (hrun : runFix p base k (s.t + 1) (view (treeSpec p) s.P s.t) = .ok (true, w))
    (hfix : fixOf p base k (s.t + 1) (view (treeSpec p) s.P s.t) = .ok (f, w1)) (hC : IsMerge f) :
    DeadAllOK p (commit s.P (s.t + 1) w) := by
  have hlift := t_liftFix_merge p s k base w f w1 hs hI hfresh hrun hfix hC
  have hkey : k ≠ [] := key_ne_nil_of_wf k (reached_pend_wf p s hs k (by rw [hpend]; exact List.mem_cons_self))
  obtain ⟨hsub, htimes, _, _, _, _⟩ := hI
  rw [hpend] at hsub
  obtain ⟨z, top, hz, htop, hfs, _, hfo⟩ := runFix_true p s.P s.t base k w hrun
  obtain ⟨x, d, rfl⟩ := hC
  have hpt := hfo _ w1 hfix
  have hroot := rootAt_of_top p s.P s.t htimes z top hz htop
  have hvc := viewCoherent_at p s.P s.t htimes
  obtain ⟨hPI, _⟩ := putInv_at p s.P s.t base hfresh htimes hsub.links hsub.tree hsub.range z top hz htop
  have hRO := rangeOrder_at p s.P s.t htimes hsub.range top.rootKey hroot base top rfl
  have hN : wNode p (viewOf s.P) (wStart base top) = knodeOf p (viewOf s.P) := wNode_wStart _ _ _ _
  have hwf0 : ∀ y yn, wNode p (viewOf s.P) (wStart base top) y = some yn → klongsWf yn.ks = true := by
    intro y yn hy
    rw [hN] at hy
    obtain ⟨nd, hg, hd, hlv, hn⟩ := knodeOf_viewOf p _ y yn hy
    subst hn
    exact (hsub.links.2.1 y nd hg hd hlv).1
  have hpath := writerWP_pathTarget k hkey (ctxAt p s.P s.t) (viewOf s.P) hvc (wStart base top)
    (fun _ _ => rfl) (fun y yn _ hy => hwf0 y yn hy) _ w1 hpt
  obtain ⟨hr1, _, _, _, _, hmg⟩ := hpath
  obtain ⟨X, K, Nn, hP, hXd, hd0, hql, _, _, _⟩ := hmg x d rfl
  -- the books at the state after the path and the target
  have hPI1 := putInv_of_readsAs _ _ _ _ hPI hr1
  have e1 : wNode p (viewOf s.P) w1 = wNode p (viewOf s.P) (wStart base top) := hr1.1
  have hRO1 : RangeOrder (wNode p (viewOf s.P) w1) (s.t + 1) w1.top.rootKey := by
    rw [e1, hr1.2.1]; exact hRO
  have hDW0 : DW (ctxAt p s.P s.t) (viewOf s.P) (wStart base top) := by
    refine ⟨?_, hwf0⟩
    intro hθ y yn hy
    rw [show (ctxAt p s.P s.t).p = p from rfl, hN] at hy
    rw [show (ctxAt p s.P s.t).p = p from rfl]
    exact hsub.dead y yn (by rw [keyNodeAt_eq_knodeOf]; exact hy) hθ
  have hDW1 := dw_of_wNode hr1.1 hDW0
  have hP' := hP
  obtain ⟨hX, _, hS2, _, _⟩ := hP'
  have hreach := pathOK_reach (ctxAt p s.P s.t) (viewOf s.P) w1 d X K Nn hP
  have hw : ∀ j, j ≤ d → wNode p (viewOf s.P) w1 (X j) = some (Nn j) := by
    intro j hj
    obtain ⟨a, b⟩ := hS2 j hj
    simp only [wNode, b, Bool.false_eq_true, ↓reduceIte, a]
  have hpar : ∀ q, (w1.path[d - 1]?).map (·.2) = some q →
      TReach (wNode p (viewOf s.P) w1) (s.t + 1) w1.top.rootKey q ∧ ∃ qn,
        wNode p (viewOf s.P) w1 q = some qn ∧ qn.leaf = false ∧
        (∀ s', Klong.time s' ∈ qn.ks → s' ≤ s.t + 1) ∧
        ∀ cc y yn, (cc, y) ∈ qn.liveAt (s.t + 1) → wNode p (viewOf s.P) w1 y = some yn →
          ∀ s', Klong.time s' ∈ yn.ks → s' ≤ s.t + 1 := by
    intro q hq
    have hqX : q = X (d - 1) := by
      rw [hX (d - 1) (by omega)] at hq; exact (Option.some.inj hq).symm
    subst hqX
    exact ⟨hreach (d - 1) (by omega), Nn (d - 1), hw (d - 1) (by omega), hql,
      hPI1.times _ _ (hw (d - 1) (by omega)), fun cc y yn _ hy => hPI1.times y yn hy⟩
  have hdw := dw_mtMerge (viewOf s.P) x d (ctxAt p s.P s.t) w1 hvc hth (by show 1 ≤ p.mergeMax; have h1 := hp.1; have h2 := hp.2.1; omega)
    hDW1 hPI1.tree hRO1 hpar (fun y _ => by rw [hr1.2.2.1]; rfl)
  obtain ⟨u, w2, _, _, m, hmm⟩ := fixStep_merge_split k _ _ w x d w1 hfs hpt
  have hDW := hdw m w hmm
  intro y n hy hθ
  rw [keyNodeAt_eq_knodeOf, hlift.knode] at hy
  exact hDW.1 hθ y n hy

/-- The kinds and the heights after a fix step of kind C: what is left of `PartsKHD`. -/
def PartsKH (p : Params) (C : Fix → Prop) : Prop :=
  ∀ (s : St) (k : Key) (rest : List Key) (base : NodeId) (w : W) (f : Fix) (w1 : W), Reached p s →
    StInv p s → s.pend = k :: rest → (∀ q ∈ s.P, q.1 < base) →
    runFix p base k (s.t + 1) (view (treeSpec p) s.P s.t) = .ok (true, w) →
    fixOf p base k (s.t + 1) (view (treeSpec p) s.P s.t) = .ok (f, w1) → C f →
    KindsAllOK p (commit s.P (s.t + 1) w) ∧ HeightAllOK p (commit s.P (s.t + 1) w) (s.t + 1)

/-- `PartsKHD` for the merge with the dead klongs proved. -/
theorem t_partsKHD_merge (p : Params) (hp : MergeParamsOK p) (hth : 2 ≤ p.theta ∨ p.theta = 0)
    (hKH : PartsKH p IsMerge) : PartsKHD p IsMerge := by
  intro s k rest base w f w1 hs hI hpend hfresh hrun hfix hC
  obtain ⟨hK, hH⟩ := hKH s k rest base w f w1 hs hI hpend hfresh hrun hfix hC
  exact ⟨hK, hH, dead_fix_merge p hp hth s k rest base w f w1 hs hI hpend hfresh hrun hfix hC⟩
