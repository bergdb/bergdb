/-! ## The user's run: a fact every operation keeps holds at its end -/

theorem userStep_ok' (op : Op) (c : Ctx) (w w1 : W) (b : Bool) (h : userStep op c w = .ok (b, w1)) :
    putKv op c w = .ok (b, w1) ∧ w1.outside = [] := by
  unfold userStep at h
  obtain ⟨noop, u, h1, h2⟩ := (m_bind_ok _ _ _ _ _).mp h
  obtain ⟨_, u', h3, h4⟩ := (m_bind_ok _ _ _ _ _).mp h2
  obtain ⟨rfl, hout⟩ := boundsNow_ok _ _ _ _ h3
  simp only [pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq, Prod.mk.injEq] at h4
  obtain ⟨rfl, rfl⟩ := h4
  exact ⟨h1, hout⟩

theorem userRun_keeps (Inv : W → Prop) (c : Ctx) :
    ∀ (ops : List Op), (∀ op ∈ ops, ∀ v, Inv v → WriterWP (putKv op) c v (fun _ v' => Inv v')) →
      ∀ (w : W) (noops : List Bool) (w' : W), Inv w → ops.mapM userStep c w = .ok (noops, w') →
        Inv w' ∧ (ops ≠ [] → w'.outside = []) := by
  intro ops
  induction ops with
  | nil =>
    intro _ w noops w' hw hrun
    simp only [List.mapM_nil, pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq,
      Prod.mk.injEq] at hrun
    obtain ⟨_, rfl⟩ := hrun
    exact ⟨hw, fun h => absurd rfl h⟩
  | cons op ops ih =>
    intro hop w noops w' hw hrun
    rw [List.mapM_cons] at hrun
    obtain ⟨b, w1, h1, h2⟩ := (m_bind_ok _ _ _ _ _).mp hrun
    obtain ⟨bs, w2, h3, h4⟩ := (m_bind_ok _ _ _ _ _).mp h2
    simp only [pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq,
      Prod.mk.injEq] at h4
    obtain ⟨_, rfl⟩ := h4
    obtain ⟨hp1, hout1⟩ := userStep_ok' op c w w1 b h1
    have hw1 := hop op List.mem_cons_self w hw b w1 hp1
    obtain ⟨hw2, hout2⟩ := ih (fun o ho => hop o (List.mem_cons_of_mem _ ho)) w1 bs w2 hw1 h3
    refine ⟨hw2, fun _ => ?_⟩
    cases ops with
    | nil =>
      simp only [List.mapM_nil, pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq,
        Prod.mk.injEq] at h3
      obtain ⟨_, rfl⟩ := h3
      exact hout1
    | cons o os => exact hout2 (by simp)

/-- The books at the end of the user's run (B1's `EndW`, as `endW_fix` for a fix step). -/
theorem endW_user (p : Params) (P : Struct BVal) (t : Time) (base : NodeId) (z : NodeId) (top : Top)
    (ops : List Op) (noops : List Bool) (w : W) (hfresh : ∀ q ∈ P, q.1 < base)
    (htimes : StructTimesLE P t) (hL : RunLinksOK p P t) (hEnc : ViewEncoded (viewOf P))
    (hM : 2 ≤ p.M) (htop : topOf p (ctxAt p P t).view z = .ok top)
    (hkw : ∀ op ∈ ops, keyWf op.key = true)
    (hrun : ops.mapM userStep (ctxAt p P t) (wStart base top) = .ok (noops, w)) :
    EndW p P (t + 1) base top w := by
  have hB := putBase_at p P t base hfresh htimes hL
  have hb0 := putBook_at p P t base hfresh htimes hL z top htop
  let c := ctxAt p P t
  have run := fun (Inv : W → Prop) (hop : ∀ op ∈ ops, ∀ v, Inv v → WriterWP (putKv op) c v
    (fun _ v' => Inv v')) (h0 : Inv (wStart base top)) => (userRun_keeps Inv c ops hop _ _ _ h0 hrun).1
  have hgood := run (CacheGood base c) (fun op _ v hv => good_mtPutKv base op c v hB.good hv) hb0.good
  have hkwf := run WriterKlongsWf (fun op _ v hv => klongsWf_mtPutKv op c v hB.wf hv) hb0.klongs
  have halloc := run ((allocatedIdsInvariant base).pred c)
    (fun op _ v hv => @state_mtPutKv (allocatedIdsInvariant base) op c v hv) hb0.alloc
  have hled := run (Ledger c) (fun op _ v hv => @state_mtPutKv ledgerInvariant op c v hv)
    (show Ledger c (wStart base top) from
      ⟨by intro nn hn; simp [wStart] at hn, by intro q hq; simp [wStart] at hq, by intro h; simp [wStart] at h⟩)
  have hbook := run (Book base c) (fun op _ v hv =>
    writerWP_mono (book_mtPutKv base op c v hB.good.ids hv) (fun _ _ h => h.1)) hb0.book
  have htk := run (fun v => v.taken = []) (fun op _ v hv => takenP_mtPutKv (fun l => l = []) op c v hv) rfl
  have hmn := run (MadeNodes c) (fun op _ v hv => @state_mtPutKv madeNodesInvariant op c v hv)
    (show MadeNodes c (wStart base top) by intro m hm; simp [wStart] at hm)
  have hcommit := run ((commitInvariant top).pred c)
    (fun op _ v hv => @state_mtPutKv (commitInvariant top) op c v hv) hb0.commit
  have hmd := run MadeData (fun op _ v hv => @state_mtPutKv madeDataInvariant op c v hv)
    (show MadeData (wStart base top) by intro m hm; simp [wStart] at hm)
  have hmk := run (MadeKinds c) (fun op _ v hv => @state_mtPutKv madeKindsInvariant op c v hv)
    (show MadeKinds c (wStart base top) by intro m hm; simp [wStart] at hm)
  have hpt := run (PtrsOK (fun y => (P.get? y).isSome = true) c) (fun op _ v hv =>
    writerWP_mono (ptrs_mtPutKv _ op c v (viewPtrsOld_at p P t htimes hL.1) hv) (fun _ _ h => h.1))
    (ptrs_start _ c base top (topOf_old p P t htimes hL.1 z top htop))
  have hne := run (fun v => CacheGood base c v ∧ Book base c v ∧ NEx [] v)
    (fun op _ v hv => neBook_mtPutKv base op c v hB.good hM hv.1 hv.2.1 hv.2.2)
    ⟨hb0.good, hb0.book, by intro q hq; simp [wStart] at hq⟩
  have hen := run WriterEncoded (fun op hop v hv => encoded_mtPutKv op
    (keyWf_pairs_encoded op.key (hkw op hop)) c v (viewEncoded_rootless p P t htimes hEnc) hv)
    (show WriterEncoded (wStart base top) from
      ⟨by intro q hq; simp [wStart] at hq, by intro q hq; simp [wStart] at hq⟩)
  have hout : w.outside = [] := by
    by_cases hops : ops = []
    · subst hops
      simp only [List.mapM_nil, pure, ReaderT.pure, StateT.pure, Except.pure, Except.ok.injEq,
        Prod.mk.injEq] at hrun
      obtain ⟨_, rfl⟩ := hrun
      rfl
    · exact (userRun_keeps (fun _ => True) c ops (fun _ _ _ _ _ _ _ => trivial) _ _ _ trivial hrun).2 hops
  exact ⟨hgood, hkwf, halloc, hled, hbook, htk, hout, hmn, hcommit, hmd, hmk, hpt,
    fun q hq h => by have := hne.2.2 q hq h; simp at this, hen⟩

/-- B1's lift after a user subtransaction, by proof. -/
theorem t_liftUser (p : Params) (hp : MergeParamsOK p) : LiftUser p := by
  intro s tx noops w hs hI hpend hsorted hwf hfresh hrun
  obtain ⟨hsub, htimes, hK, hD, hEnc, _⟩ := hI
  obtain ⟨z, top, hz, htop, hmap⟩ := runUser_ok p s.P s.t tx.base tx.ops noops w hrun
  have hroot0 := rootAt_of_top p s.P s.t htimes z top hz htop
  have hkw : ∀ op ∈ tx.ops, keyWf op.key = true := by
    intro op hop
    have := List.all_eq_true.mp hwf op hop
    simpa using this
  have hM : 2 ≤ p.M := by have := hp.2.2.2.1; omega
  exact liftOK_of_end p s.P s.t tx.base top w hfresh htimes hsub.links hD hK hEnc (reached_root p s hs)
    hroot0 (endW_user p s.P s.t tx.base z top tx.ops noops w hfresh htimes hsub.links hEnc hM htop hkw
      hmap)

/-- The top with the user run's lift proved: what is left are the full nodes' cascades, the puts
with holes, the two frames and the user run's parts. -/
theorem t_visible_userL (p : Params) :
    MergeParamsOK p → p.merges = true → (2 ≤ p.theta ∨ p.theta = 0) →
    PutWH p → DelWFull p → FrameUser p → PartsUser p →
    MergeFullLive p → MergeFullEmpty p → FixWUnlinkFull p → FixWCollapseSeg p →
    FrameW p → VisibleOK p :=
  fun hp hmg hth hPut hDF hFU hPa hM hE hUf hSeg hFr =>
    t_visible_user p hp hmg hth hPut hDF hFU hPa (t_liftUser p hp) hM hE hUf hSeg hFr
