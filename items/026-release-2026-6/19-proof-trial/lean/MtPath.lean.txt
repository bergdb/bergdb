/-- What `target` chose, if a merge: x is the path's node at depth d ≥ 1, cached with
fewer than m live klongs; the node above is inner and cached with two or more. -/
def MergeAt (c : Ctx) (v : W) (x : NodeId) (d : Nat) : Prop :=
  1 ≤ d ∧ (v.path[d]?).map (·.2) = some x ∧ ∃ q qn xn, (v.path[d - 1]?).map (·.2) = some q ∧
    v.nodes.lookup q = some qn ∧ qn.leaf = false ∧ 2 ≤ qn.cnt ∧
    v.nodes.lookup x = some xn ∧ xn.cnt < c.p.mergeBelow

theorem pathCached_lookup {v : W} (hc : PathCached v) {d : Nat} {e : Option Nat × NodeId}
    (he : v.path[d]? = some e) : ∃ n, v.nodes.lookup e.2 = some n :=
  hc e (List.mem_of_getElem? he)

theorem writerWP_targetGo (c : Ctx) :
    ∀ (k : Nat) (v : W), PathCached v →
      WriterWP (targetGo k) c v (fun f v' => stripW v' = stripW v ∧
        ∀ x d, f = .merge x d → MergeAt c v x d) := by
  intro k
  induction k with
  | zero =>
    intro v _
    show WriterWP (pure Fix.none) c v _
    rw [writerWP_pure]
    exact ⟨rfl, fun x d h => by cases h⟩
  | succ d ih =>
    intro v hc
    unfold targetGo
    simp only [writerWP_bind, writerWP_read]
    unfold pathNode
    simp only [writerWP_bind, writerWP_get]
    split
    · rename_i e he
      obtain ⟨xn, hxn⟩ := pathCached_lookup hc he
      simp only [writerWP_pure]
      rw [writerWP_getNode_known e.2 xn c v hxn]
      have hno : ∀ f, (∀ x d, f ≠ Fix.merge x d) → WriterWP (pure f : M Fix) c v
          (fun f v' => stripW v' = stripW v ∧ ∀ x d, f = .merge x d → MergeAt c v x d) := by
        intro f hf
        rw [writerWP_pure]
        exact ⟨rfl, fun x d h => absurd h (hf x d)⟩
      have hrec := ih v hc
      have hnote : ∀ s, WriterWP (do note s; targetGo d) c v
          (fun f v' => stripW v' = stripW v ∧ ∀ x d, f = .merge x d → MergeAt c v x d) := by
        intro s
        simp only [note, writerWP_bind, writerWP_modify]
        refine writerWP_mono (ih _ hc) ?_
        intro f v' ⟨h1, h2⟩
        exact ⟨h1, fun x d hf => h2 x d hf⟩
      have hroot : WriterWP (if xn.leaf = true then
            (if (xn.cnt == 0 && d != 0) = true then pure (Fix.unlink d) else targetGo d)
          else if (xn.cnt == 1) = true then pure (Fix.collapse e.snd d) else targetGo d) c v
          (fun f v' => stripW v' = stripW v ∧ ∀ x d, f = .merge x d → MergeAt c v x d) := by
        split
        · split
          · exact hno _ (fun x d h => by cases h)
          · exact hrec
        · split
          · exact hno _ (fun x d h => by cases h)
          · exact hrec
      by_cases hemp : (!xn.leaf && xn.cnt == 0) = true
      · simp only [hemp, ↓reduceIte, writerWP_bind]; exact writerWP_throw _ _ _ _
      simp only [hemp, Bool.false_eq_true, ↓reduceIte]
      cases d with
      | zero =>
        dsimp only
        rw [writerWP_bind, writerWP_pure]
        exact hroot
      | succ e1 =>
        simp only [writerWP_bind, writerWP_get]
        split
        · rename_i e' he'
          obtain ⟨qn, hqn⟩ := pathCached_lookup hc he'
          simp only [writerWP_pure]
          rw [writerWP_getNode_known e'.2 qn c v hqn]
          try simp only [writerWP_pure]
          by_cases hql : qn.leaf = true
          · simp only [hql, ↓reduceIte]; exact hroot
          · have hql' : qn.leaf = false := by simpa using hql
            simp only [hql', Bool.false_eq_true, ↓reduceIte]
            split
            · rename_i hxc
              split
              · rename_i hqc
                rw [writerWP_pure]
                refine ⟨rfl, fun x d hf => ?_⟩
                cases hf
                exact ⟨by omega, by rw [he]; rfl, e'.2, qn, xn, by rw [show e1 + 1 - 1 = e1 by omega, he']; rfl,
                  hqn, hql', hqc, hxn, hxc⟩
              · exact hnote _
            · exact hrec
        · exact writerWP_throw _ _ _ _
    · exact writerWP_throw _ _ _ _

theorem pathOK_mono {c : Ctx} {w : W} {d D : Nat} {X : Nat → NodeId} {K : Nat → Nat}
    {Nn : Nat → KNode} (h : PathOK c w D X K Nn) (hd : d ≤ D) : PathOK c w d X K Nn := by
  obtain ⟨h1, h2, h3, h4, h5⟩ := h
  exact ⟨fun i hi => h1 i (by omega), fun i h0 hi => h2 i h0 (by omega),
    fun i hi => h3 i (by omega), fun i h0 hi => h4 i h0 (by omega), h5⟩

/-- The fix step's path and target (A2): from w, after `path key; target`, the
state reads as w, made nothing, and if the target is `merge x d`, the path down to
depth d is `PathOK` with x at depth d, d ≥ 1, the node above inner with two live
klongs or more and x with fewer than m; every node of the path is one that a
reading of the key visits. -/
theorem writerWP_pathTarget (key : Key) (hkey : key ≠ []) (c : Ctx) (base : View BVal)
    (hvc : ViewCoherent c base) (w : W)
    (hRt : ∀ y, TReach (wNode c.p base w) c.i w.top.rootKey y → w.taken.contains y = false)
    (hwf : ∀ y yn, TReach (wNode c.p base w) c.i w.top.rootKey y → wNode c.p base w y = some yn →
      klongsWf yn.ks = true) :
    WriterWP (do Mt.path key; target : M Fix) c w (fun f v => ReadsAs c.p base w v ∧
      w.nodes <+: v.nodes ∧ v.made = w.made ∧ PathCached v ∧
      (∀ e ∈ v.path, ∃ cv, (e.2, cv) ∈ fromVisits (wNode c.p base w) c.i c.p.fuel key w.top.rootKey) ∧
      ∀ x d, f = .merge x d → ∃ X K Nn, PathOK c v d X K Nn ∧ X d = x ∧ 0 < d ∧
        (Nn (d - 1)).leaf = false ∧ 2 ≤ (Nn (d - 1)).cnt ∧ (Nn d).cnt < c.p.mergeBelow ∧
        TReach (wNode c.p base w) c.i w.top.rootKey x) := by
  rw [writerWP_bind]
  refine writerWP_mono (writerWP_mtPath key hkey c base hvc w hRt hwf) ?_
  rintro _ v1 ⟨hr1, hpre1, hm1, ps, hp1, hlf1, hc1, hfv1⟩
  unfold target pathSize
  simp only [writerWP_bind, writerWP_get, writerWP_pure]
  refine writerWP_mono (writerWP_targetGo c _ v1 hc1) ?_
  rintro f v ⟨hs, hmg⟩
  obtain ⟨hn, hg, hmd, hv, htk, hnx, hpth, htop, _, _, _, _⟩ := strip_fields hs
  have hrv : ReadsAs c.p base v1 v :=
    ⟨wNode_strip c.p base hs, htop, htk, hv, hnx⟩
  have hr : ReadsAs c.p base w v := ReadsAs.trans _ _ _ _ _ hr1 hrv
  have hc : PathCached v := by
    intro e he; rw [hpth] at he; obtain ⟨m, hm⟩ := hc1 e he; exact ⟨m, by rw [hn]; exact hm⟩
  have hp : v.path = (none, w.top.rootKey) :: ps := by rw [hpth, hp1]
  refine ⟨hr, by rw [hn]; exact hpre1, by rw [hmd, hm1], hc, ?_, ?_⟩
  · intro e he
    rw [hp] at he
    exact hfv1 e.2 (by
      rcases List.mem_cons.mp he with he | he
      · rw [he]; exact List.mem_cons_self
      · exact List.mem_cons_of_mem _ (List.mem_map_of_mem he))
  · intro x d hf
    obtain ⟨hd1, hpx, q, qn, xn, hpq, hlq, hql, hqc, hlx, hxc⟩ := hmg x d hf
    rw [← hn] at hlq hlx
    have hPO := pathOK_of_linked c base w v ps hr hp hlf1 hc hRt
    have hdl : d ≤ ps.length := by
      have : d < ((none, w.top.rootKey) :: ps).length := by
        rw [← hp1]
        cases h : v1.path[d]? with
        | none => rw [h] at hpx; cases hpx
        | some _ => exact (List.getElem?_eq_some_iff.mp h).1
      simp at this; omega
    have hPd := pathOK_mono hPO hdl
    refine ⟨_, _, _, hPd, ?_, by omega, ?_, ?_, ?_, ?_⟩
    · have := hPd.1 d (Nat.le_refl _)
      rw [hpth, hpx] at this
      exact (Option.some.inj this).symm
    · have hX := hPd.1 (d - 1) (by omega)
      rw [hpth, hpq] at hX
      have hlk := (hPd.2.2.1 (d - 1) (by omega)).1
      rw [← Option.some.inj hX, hlq] at hlk
      have e := Option.some.inj hlk
      dsimp only at e ⊢
      rw [← e]; exact hql
    · have hX := hPd.1 (d - 1) (by omega)
      rw [hpth, hpq] at hX
      have hlk := (hPd.2.2.1 (d - 1) (by omega)).1
      rw [← Option.some.inj hX, hlq] at hlk
      have e := Option.some.inj hlk
      dsimp only at e ⊢
      rw [← e]; exact hqc
    · have hX := hPd.1 d (Nat.le_refl _)
      rw [hpth, hpx] at hX
      have hlk := (hPd.2.2.1 d (Nat.le_refl _)).1
      rw [← Option.some.inj hX, hlx] at hlk
      have e := Option.some.inj hlk
      dsimp only at e ⊢
      rw [← e]; exact hxc
    · have hX := hPd.1 d (Nat.le_refl _)
      rw [hpth, hpx] at hX
      rw [Option.some.inj hX]
      have := pathOK_reach c base v d _ _ _ hPd d (Nat.le_refl _)
      rw [hr.1, hr.2.1] at this
      exact this

/-- A state that reads as a fix step's first state and made nothing is one again. -/
theorem fixStart_of_reads {c : Ctx} {base : View BVal} {w v : W} (h : FixStart c base w)
    (hr : ReadsAs c.p base w v) (hm : v.made = w.made) : FixStart c base v := by
  obtain ⟨h1, h2, h3⟩ := h
  refine ⟨by rw [hm, h1], by rw [hr.2.2.1, h2], fun y yn hy => h3 y yn ?_⟩
  rw [← hr.1]; exact hy
