/-
The writer's tree, `PutInv`'s main parts and the walk's ranges after every
operation, checked on runs (bergdb 024-12, evidence for the top theorem's
premises t4, i4 and r4, the delete with the merges (4)). Not part of the Lean
project; run it from lean/:

  lake env lean ../items/024-release-2026-5/12-tree-model/delete-posts-check.lean.txt

At every state between the operations of every transaction of a run within
reach, with the merges on (record format 7's thresholds, theta 0 and Java's
theta), the writer's tree from its space root is a tree (`TreeOK`: the root a
key node, linked by no node reached, every other node reached named by one
link), the nodes reached are not taken back, have well-formed klongs of no
later time and name ids below the next id (`PutInv`'s global and reach parts),
and every node reached keeps the compare values of its live klongs in the
range its parent's klongs give (`RangeOrder`, the walk's ranges). Each state
notes its operation (put or del) and ok or FAIL. The tree and range walks are
putinv's (`UpChk.lean`, 2026-11-14).
-/

def withinB (p : Params) (all : List Tx) : Bool :=
  all.all (fun t => t.sorted && t.wf) &&
  (List.range all.length).all (fun j =>
    match all[j]? with
    | none => false
    | some tj => (treeRun p (all.take j)).all (fun q => decide (q.1 < tj.base)) &&
        isOk (exec? p tj (j + 1) (view (treeSpec p) (treeRun p (all.take j)) j)))

def putsAB (as bs : List Nat) : List Op := (as.flatMap (fun a => bs.map (fun b => .put [a, b] (a * 1000 + b + 1))))
def dels (as bs : List Nat) : List Op := (as.flatMap (fun a => bs.map (fun b => .del [a, b])))
def A (n : Nat) : List Nat := (List.range n).map (fun i => 64 * i + 56)
def Bs (n : Nat) : List Nat := (List.range n).map (fun j => 64 * j + 8)
def mkRun (na nb k1 keep : Nat) : List Tx :=
  let a0 := 64 * (na / 2) + 56
  [{base := 1000, ops := putsAB (A na) (Bs 2)},
   {base := 3000, ops := putsAB [a0] ((Bs nb).drop 2)},
   {base := 5000, ops := dels [a0] ((Bs nb).take k1)},
   {base := 7000, ops := dels [a0] (((Bs nb).drop k1).take (nb - k1 - keep))}]
def mkRun1 (n k : Nat) : List Tx :=
  [{base := 1000, ops := (List.range n).map (fun j => .put [64*j+8] (j+1))},
   {base := 3000, ops := (List.range k).map (fun j => .del [64*(2*j)+8])},
   {base := 5000, ops := (List.range n).map (fun j => .del [64*j+8])}]
def family : List (List Tx) :=
  [1, 3, 7].flatMap (fun na => [12, 20, 30].flatMap (fun nb =>
    [nb / 2, nb - 8, nb - 6].flatMap (fun k1 => [1, 2].map (fun keep => mkRun na nb k1 keep))))
def family1 : List (List Tx) :=
  [10, 25, 40].flatMap (fun n => [3, 8, 15].map (fun k => mkRun1 n k))

def mkTx (b : Nat) (ops : List Op) : Tx := {base := b, ops := ops}
def keysN (n : Nat) : List (List Nat) := (List.range n).map (fun j => [64 * j + 8])
def roundOps (ks : List (List Nat)) (f : Nat → Option Bool) (v : Nat) : List Op :=
  (ks.zipIdx.filterMap (fun (key, j) => match f j with
    | some true => some (.put key v)
    | some false => some (.del key)
    | none => none))
def mkOver (n r k : Nat) : List Tx :=
  let ks := keysN n
  [mkTx 1000 (roundOps ks (fun _ => some true) 1)] ++
  (List.range r).map (fun i => mkTx (3000 + 2000 * i) (roundOps ks (fun j =>
    if n - k ≤ j ∧ i % 2 = 0 then some false
    else if j < n - i % 3 then some true else none) (i + 2))) ++
  [mkTx (3000 + 2000 * r) (roundOps ks (fun j => if j < k then some false else none) 0)]
/-- Overwrites in two segments. -/
def mkOver2 (na nb r : Nat) : List Tx :=
  let ks := (A na).flatMap (fun a => (Bs nb).map (fun b => [a, b]))
  let m := ks.length
  [mkTx 1000 (roundOps ks (fun _ => some true) 1)] ++
  (List.range r).map (fun i => mkTx (5000 + 4000 * i) (roundOps ks (fun j =>
    if m / 3 ≤ j ∧ j < m / 3 + i + 1 then some false
    else if i ≤ j ∧ j < i + m / 2 then some true else none) (i + 2)))
def familyO : List (List Tx) :=
  [4, 7, 12, 20].flatMap (fun n => [3, 6].flatMap (fun r => [1, 3].map (fun k => mkOver n r k)))
def familyO2 : List (List Tx) :=
  [2, 3].flatMap (fun na => [4, 9].flatMap (fun nb => [3, 5].map (fun r => mkOver2 na nb r)))
def dP8 : Params := ({M := 8, G := 1, H := 2} : Params).withMerges.withTheta
def dP10 : Params := ({M := 10, G := 2, H := 2} : Params).withMerges.withTheta

def dP30 : Params := ({M := 30, G := 1, H := 2} : Params).withMerges.withTheta
def linksB (N : NodeId → Option KNode) (t : Time) (u : NodeId) : List (Nat × NodeId) :=
  match N u with
  | some n => n.liveAt t
  | none => []

def reachB (N : NodeId → Option KNode) (t : Time) : Nat → List NodeId → List NodeId → List NodeId
  | 0, _, acc => acc
  | _ + 1, [], acc => acc
  | f + 1, u :: todo, acc =>
    if acc.contains u then reachB N t f todo acc
    else reachB N t f (todo ++ (linksB N t u).map (·.2)) (u :: acc)

def treeOKB (N : NodeId → Option KNode) (t : Time) (r : NodeId) : Bool :=
  let R := reachB N t 100000 [r] []
  (N r).isSome &&
  R.all (fun u => (linksB N t u).all (fun q => q.2 != r)) &&
  R.all (fun y => y == r ||
    ((R.map (fun u => ((linksB N t u).filter (fun q => q.2 == y)).length)).foldl (· + ·) 0) == 1)

def nodeRangeB (n : KNode) (t lo : Nat) (hi : Option Nat) : Bool :=
  (n.liveAt t).all (fun q => (if n.leaf then decide (lo ≤ q.1) else (q.1 == 0 || decide (lo < q.1))) &&
    (match hi with | none => true | some h => decide (q.1 < h)))

def rangeWalkB (N : NodeId → Option KNode) (t : Time) : Nat → List (NodeId × Nat × Option Nat) → Bool
  | 0, _ => false
  | _ + 1, [] => true
  | f + 1, (x, lo, hi) :: todo =>
    match N x with
    | none => rangeWalkB N t f todo
    | some n =>
      let kids : List (NodeId × Nat × Option Nat) :=
        if n.leaf then (n.liveAt t).map (fun q => (q.2, 0, none))
        else (childBounds (n.liveAt t) hi).map (fun e => (e.1.2, if e.1.1 == 0 then lo else e.1.1, e.2))
      nodeRangeB n t lo hi && rangeWalkB N t f (todo ++ kids)


/-- The checks at one writer's state. -/
def postsB (p : Params) (P : Struct BVal) (t : Time) (w : W) : Bool × Bool × Bool :=
  let N := wNode p (viewOf P) w
  let R := reachB N t 100000 [w.top.rootKey] []
  (treeOKB N t w.top.rootKey,
   R.all (fun y => !w.taken.contains y && (match N y with
     | some n => klongsWf n.ks && (n.ks.all (fun k => match k with
         | .time s => decide (s ≤ t)
         | _ => true)) && (n.liveAt t).all (fun q => decide (q.2 < w.next))
     | none => true)),
   rangeWalkB N t 100000 [(w.top.rootKey, 0, none)])

/-- One note per state after an operation: its kind and the three checks. -/
def runPosts (p : Params) (all : List Tx) : List String :=
  (if withinB p all then [] else ["NOT-WITHIN"]) ++
  (List.range all.length).foldl (fun acc j =>
    match all[j]? with
    | none => acc
    | some tx =>
      let ts := all.take j
      let v := view (treeSpec p) (treeRun p ts) ts.length
      match lastTop v with
      | none => acc ++ ["notop"]
      | some z =>
        match topOf p (ctxOf p ts).view z with
        | .error _ => acc ++ ["toperr"]
        | .ok top =>
          acc ++ (List.range tx.ops.length).map (fun k =>
            let kind := match tx.ops[k]? with
              | some (.del _) => "del"
              | _ => "put"
            match ((tx.ops.take (k + 1)).mapM putKv) (ctxOf p ts) (wInit tx top) with
            | .ok (_, w) =>
              let r := postsB p (treeRun p ts) (ts.length + 1) w
              kind ++ ":" ++ (if r.1 then "t" else "T") ++ (if r.2.1 then "i" else "I") ++
                (if r.2.2 then "r" else "R")
            | .error e => "ERR " ++ e)) []

def summP (l : List String) : String :=
  let cnt (s : String) := (l.filter (· == s)).length
  let bad := l.filter (fun s => !(s == "del:tir" || s == "put:tir"))
  s!"states {l.length} del ok {cnt "del:tir"} put ok {cnt "put:tir"} other {bad.length} {bad.take 3}"
