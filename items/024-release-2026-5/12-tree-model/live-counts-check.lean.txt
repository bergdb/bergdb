/-
LiveCounts, checked on runs (bergdb 024-12, TP7, `LiveCounts.lean`). Not part
of the Lean project; run it from lean/:

  lake env lean ../items/024-release-2026-5/12-tree-model/live-counts-check.lean.txt

At every state between the operations of every transaction of a run within
reach, with the merges on (record format 7's thresholds), the writer's tree
from its space root meets `LiveCounts`: a live klong in every node reached
but the space root as a segment leaf, two in an inner segment root, m in a
node an inner node names. Each state notes lc:ok or lc:FAIL.
-/
import SIS.Tree.MergeProofBundle

set_option autoImplicit false
set_option maxRecDepth 100000

namespace SIS.Tree

open SIS SIS.Check

/-- The nodes reached with the walk's flags (`FReach`), breadth first. -/
def lcNodes (N : NodeId → Option KNode) (t : Time) :
    Nat → List (NodeId × Bool × Bool) → List (NodeId × Bool × Bool) → List (NodeId × Bool × Bool)
  | 0, _, acc => acc
  | _ + 1, [], acc => acc
  | f + 1, q :: todo, acc =>
    if acc.contains q then lcNodes N t f todo acc
    else
      match N q.1 with
      | none => lcNodes N t f todo (q :: acc)
      | some n => lcNodes N t f (todo ++ (n.liveAt t).map (fun e =>
          if n.leaf then (e.2, true, false) else (e.2, false, q.2.2))) (q :: acc)

def lcB (p : Params) (N : NodeId → Option KNode) (t : Time) (r : NodeId) : Bool :=
  (lcNodes N t 100000 [(r, true, true)] []).all (fun q =>
    match N q.1 with
    | none => true
    | some n =>
      let len := (n.liveAt t).length
      (!(n.liveAt t).isEmpty || (q.2.1 && q.2.2 && n.leaf)) &&
      (n.leaf || !q.2.1 || decide (2 ≤ len)) &&
      (!p.merges || q.2.1 || decide (p.mergeBelow ≤ len)))

def withinB (p : Params) (all : List Tx) : Bool :=
  all.all (fun t => t.sorted && t.wf) &&
  (List.range all.length).all (fun j =>
    match all[j]? with
    | none => false
    | some tj => (treeRun p (all.take j)).all (fun q => decide (q.1 < tj.base)) &&
        isOk (exec? p tj (j + 1) (view (treeSpec p) (treeRun p (all.take j)) j)))

/-- One note per state between operations: lc:ok or lc:FAIL, and the number of
nodes reached. -/
def runLc (p : Params) (all : List Tx) : List String :=
  (if withinB p all then [] else ["NOT-WITHIN"]) ++
  (List.range all.length).foldl (fun acc j =>
    match all[j]? with
    | none => acc
    | some tx =>
      let ts := all.take j
      let v := view (treeSpec p) (treeRun p ts) ts.length
      match lastTop v with
      | none => acc ++ ["notop"]
      | some z =>
        match topOf p (ctxOf p ts).view z with
        | .error _ => acc ++ ["toperr"]
        | .ok top =>
          acc ++ (List.range (tx.ops.length + 1)).map (fun k =>
            match ((tx.ops.take k).mapM putKv) (ctxOf p ts) (wInit tx top) with
            | .ok (_, w) =>
              if lcB p (wNode p (viewOf (treeRun p ts)) w) (ts.length + 1) w.top.rootKey
              then "lc:ok" else "lc:FAIL"
            | .error e => "ERR " ++ e)) []

def summ (l : List String) : String :=
  let cnt (s : String) := (l.filter (· == s)).length
  s!"states {l.length} ok {cnt "lc:ok"} FAIL {cnt "lc:FAIL"} other {(l.filter (fun s => s.startsWith "ERR" || s.startsWith "NOT" || s.startsWith "not" || s.startsWith "toperr")).take 3}"

def putsAB (as bs : List Nat) : List Op := (as.flatMap (fun a => bs.map (fun b => .put [a, b] (a * 1000 + b + 1))))
def dels (as bs : List Nat) : List Op := (as.flatMap (fun a => bs.map (fun b => .del [a, b])))
def A (n : Nat) : List Nat := (List.range n).map (fun i => 64 * i + 56)
def Bs (n : Nat) : List Nat := (List.range n).map (fun j => 64 * j + 8)
def mkRun (na nb k1 keep : Nat) : List Tx :=
  let a0 := 64 * (na / 2) + 56
  [{base := 1000, ops := putsAB (A na) (Bs 2)},
   {base := 3000, ops := putsAB [a0] ((Bs nb).drop 2)},
   {base := 5000, ops := dels [a0] ((Bs nb).take k1)},
   {base := 7000, ops := dels [a0] (((Bs nb).drop k1).take (nb - k1 - keep))}]
def mkRun1 (n k : Nat) : List Tx :=
  [{base := 1000, ops := (List.range n).map (fun j => .put [64*j+8] (j+1))},
   {base := 3000, ops := (List.range k).map (fun j => .del [64*(2*j)+8])},
   {base := 5000, ops := (List.range n).map (fun j => .del [64*j+8])}]
def family : List (List Tx) :=
  [1, 3, 7].flatMap (fun na => [12, 20, 30].flatMap (fun nb =>
    [nb / 2, nb - 8, nb - 6].flatMap (fun k1 => [1, 2].map (fun keep => mkRun na nb k1 keep))))
def family1 : List (List Tx) :=
  [10, 25, 40].flatMap (fun n => [3, 8, 15].map (fun k => mkRun1 n k))
def lcP8 : Params := ({M := 8, G := 1, H := 2} : Params).withMerges
def lcP10 : Params := ({M := 10, G := 2, H := 2} : Params).withMerges
def lcP14 : Params := ({M := 14, G := 2, H := 3} : Params).withMerges
#eval (lcP8, lcP10, lcP14)
#eval summ (family1.flatMap (fun r => runLc lcP8 r))
#eval summ (family.flatMap (fun r => runLc lcP8 r))
#eval summ (family.flatMap (fun r => runLc lcP10 r))
#eval summ (family1.flatMap (fun r => runLc lcP14 r))
end SIS.Tree
